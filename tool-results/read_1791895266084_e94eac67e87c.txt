   159→   159→  private buildWorld(): void {
   160→   160→    const S = WORLD.SIZE;
   161→   161→    const m = WORLD.GRASS_MARGIN;
   162→   162→    // Вода (фон)

   172→   172→    const foamStep = 192;
   173→   173→    const foamAnim = this.anims.get('fx-foam');
   174→   174→    const foamRing = (x: number, y: number) => {
   175→   175→      const f = this.add.sprite(x, y, 'fx-foam').setDepth(DEPTH.WATER + 2);
   176→   176→      f.play('fx-foam');
   177→   177→      if (foamAnim) f.anims.setCurrentFrame(foamAnim.frames[Phaser.Math.Between(0, foamAnim.frames.length - 1)]);
   178→   178→    };
   179→   179→    for (let x = m; x < S - m; x += foamStep) {
   180→   180→      foamRing(x + foamStep / 2, m);
   181→   181→      foamRing(x + foamStep / 2, S - m);
   182→   182→    }
   183→   183→    for (let y = m; y < S - m; y += foamStep) {
   184→   184→      foamRing(m, y + foamStep / 2);
   185→   185→      foamRing(S - m, y + foamStep / 2);
   186→   186→    }
   187→   187→
   188→   188→    this.physics.world.setBounds(m, m, S - m * 2, S - m * 2);
   189→   189→
   190→   190→    // Детерминированный декор
   191→   191→    const rng = new Phaser.Math.RandomDataGenerator(['tiny-sword-survivors']);
   192→   192→    const cx = S / 2, cy = S / 2;
   193→   193→    const placed: { x: number; y: number; r: number }[] = [];
   194→   194→    const canPlace = (x: number, y: number, r: number): boolean => {
   195→   195→      if (Math.hypot(x - cx, y - cy) < 380) return false; // зона спавна
   196→   196→      for (const p of placed) {
   197→   197→        if (Math.hypot(x - p.x, y - p.y) < p.r + r) return false;
   198→   198→      }
   199→   199→      return true;
   200→   200→    };
   201→   201→    const place = (x: number, y: number, r: number) => placed.push({ x, y, r });
   202→   202→    const randPos = (margin = 260): [number, number] => [
   203→   203→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   204→   204→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   205→   205→    ];
   206→   206→
   207→   207→    // Замок — дом игрока
   208→   208→    const castle = this.add.image(cx, cy - 460, 'castle').setScale(0.9)
   209→   209→      .setDepth(DEPTH.ENTITY_BASE + cy - 460);
   210→   210→    void castle;
   211→   211→    place(cx, cy - 460, 200);
   212→   212→    this.add.image(cx - 240, cy - 420, 'house2').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 420);
   213→   213→    this.add.image(cx + 230, cy - 430, 'house1').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 430);

   215→   215→    this.add.image(cx + 320, cy - 280, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 280);
   216→   216→    this.add.image(cx + 60, cy + 470, 'monastery').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy + 470);
   217→   217→    place(cx + 60, cy + 470, 160);
   218→   218→
   219→   219→    // Дома и башни по карте
   220→   220→    for (let i = 0; i < 8; i++) {
   221→   221→      const [x, y] = randPos();
   222→   222→      if (canPlace(x, y, 120)) {
   223→   223→        const k = Phaser.Math.RND.pick(['house1', 'house2', 'house3']);
   224→   224→        this.add.image(x, y, k).setScale(0.75).setDepth(DEPTH.ENTITY_BASE + y);
   225→   225→        place(x, y, 120);
   226→   226→      }
   227→   227→    }
   228→   228→    for (let i = 0; i < 6; i++) {
   229→   229→      const [x, y] = randPos();
   230→   230→      if (canPlace(x, y, 110)) {
   231→   231→        this.add.image(x, y, 'tower').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + y);
   232→   232→        place(x, y, 110);
   233→   233→      }
   234→   234→    }
   235→   235→    // Деревья
   236→   236→    for (let i = 0; i < 70; i++) {
   237→   237→      const [x, y] = randPos();
   238→   238→      if (!canPlace(x, y, 80)) continue;
   239→   239→      const key = rng.frac() < 0.6 ? 'tree1' : 'tree3';
   240→   240→      const sc = 0.8 + rng.frac() * 0.35;
   241→   241→      const t = this.add.sprite(x, y, key).setScale(sc);
   242→   242→      t.play(key === 'tree1' ? 'tree1-sway' : 'tree3-sway');
   243→   243→      if (t.anims.currentAnim) t.setFrame(rng.between(0, 7));
   244→   244→      t.setDepth(DEPTH.ENTITY_BASE + y);
   245→   245→      place(x, y, 80);
   246→   246→    }
   247→   247→    // Камни и кусты
   248→   248→    for (let i = 0; i < 30; i++) {
   249→   249→      const [x, y] = randPos();
   250→   250→      if (!canPlace(x, y, 40)) continue;
   251→   251→      this.add.image(x, y, `rock${rng.between(1, 3)}`).setScale(0.8 + rng.frac() * 0.5)
   252→   252→        .setDepth(DEPTH.ENTITY_BASE + y - 4);
   253→   253→      place(x, y, 40);
   254→   254→    }
   255→   255→    for (let i = 0; i < 40; i++) {
   256→   256→      const [x, y] = randPos();
   257→   257→      if (!canPlace(x, y, 40)) continue;
   258→   258→      const b = this.add.sprite(x, y, 'bushe1').setScale(0.7 + rng.frac() * 0.4);
   259→   259→      b.play('bushe1-sway');
   260→   260→      b.setFrame(rng.between(0, 7));
   261→   261→      b.setDepth(DEPTH.ENTITY_BASE + y - 2);
   262→   262→      place(x, y, 40);
   263→   263→    }
   264→   264→    // Овечки
   265→   265→    for (let i = 0; i < 10; i++) {
   266→   266→      const [x, y] = randPos();
   267→   267→      if (!canPlace(x, y, 50)) continue;
   268→   268→      const sh = this.add.sprite(x, y, 'sheep').setScale(0.8);
   269→   269→      sh.play('sheep-idle');
   270→   270→      sh.setDepth(DEPTH.ENTITY_BASE + y);
   271→   271→      place(x, y, 50);
   272→   272→    }
   273→   273→    // Облака (дрейф вправо, относительный твин с переносом)
   274→   274→    for (let i = 0; i < 10; i++) {
   275→   275→      const [x, y] = [rng.between(0, S), rng.between(0, S)];
   276→   276→      const c = this.add.image(x, y, `cloud${rng.between(1, 3)}`)
   277→   277→        .setAlpha(0.5)
   278→   278→        .setDepth(DEPTH.FX + 50);
   279→   279→      this.tweens.add({
   280→   280→        targets: c,
   281→   281→        x: '+=700',
   282→   282→        duration: rng.between(60, 100) * 1000,
   283→   283→        repeat: -1,
   284→   284→        onRepeat: () => { c.x = -320; },
   285→   285→      });
   286→   286→    }
   287→   287→  }

   347→   347→  private setupCollisions(): void {
   348→   348→    this.physics.add.overlap(this.player, this.enemyGroup, (_p, e) => {
   349→   349→      const enemy = e as Enemy;
   350→   350→      this.hurtPlayer(enemy.damage);
   351→   351→    });
   352→   352→
   353→   353→    this.physics.add.overlap(this.projectiles.arrows, this.enemyGroup, (proj, e) => {
   354→   354→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   355→   355→      const enemy = e as Enemy;
   356→   356→      if (!p.active || !enemy.active) return;
   357→   357→      if (p.hitSet?.has(enemy)) return;
   358→   358→      p.hitSet?.add(enemy);
   359→   359→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 90);
   360→   360→      this.projectiles.consumePierce(p);
   361→   361→    });
   362→   362→
   363→   363→    this.physics.add.overlap(this.projectiles.spears, this.enemyGroup, (proj, e) => {
   364→   364→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   365→   365→      const enemy = e as Enemy;
   366→   366→      if (!p.active || !enemy.active) return;
   367→   367→      if (p.hitSet?.has(enemy)) return;
   368→   368→      p.hitSet?.add(enemy);
   369→   369→      this.damageEnemy(enemy, p.dmg, 0, 0, 40);
   370→   370→      this.projectiles.consumePierce(p);
   371→   371→    });
   372→   372→
   373→   373→    this.physics.add.overlap(this.projectiles.knives, this.enemyGroup, (proj, e) => {
   374→   374→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   375→   375→      const enemy = e as Enemy;
   376→   376→      if (!p.active || !enemy.active) return;
   377→   377→      if (p.hitSet?.has(enemy)) return;
   378→   378→      p.hitSet?.add(enemy);
   379→   379→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 70);
   380→   380→      this.projectiles.consumePierce(p);
   381→   381→    });
   382→   382→
   383→   383→    this.physics.add.overlap(this.player, this.projectiles.enemyArrows, (_p, proj) => {
   384→   384→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   385→   385→      if (!p.active) return;
