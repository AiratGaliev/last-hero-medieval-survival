   106→  create(payload?: { hero?: HeroId; diff?: DifficultyId }): void {
   107→    this.resetState();
   108→    this.initHero(payload?.hero);
   109→    this.initDifficulty(payload?.diff);
   110→    this.shopLevels = getShopLevels();
   111→    this.buildWorld();
   112→    this.buildPlayer();
   113→    // Зум камеры ДО создания HUD: HUD читает зум при постройке и
   114→    // компенсирует его (иначе первый кадр рисует HUD не на месте).
   115→    this.resizeZoom();
   116→    this.scale.on('resize', this.resizeZoom, this);
   117→    this.buildSystems();
   118→    this.recomputeStats();
   119→    this.setupCollisions();
   120→    this.setupInput();
   121→    this.cameras.main.setBackgroundColor('#3a7bd5');
   122→    this.cameras.main.setBounds(0, 0, WORLD.SIZE, WORLD.SIZE);
   123→    this.cameras.main.startFollow(this.player, true, 0.12, 0.1);
   124→    // Амбиент суток: MULTIPLY-прямоугольник над миром, но ПОД HUD — смена
   125→    // освещения от полудня к ночи за 15 минут (финальный босс — ночью)
   126→    this.ambientRect = this.add.rectangle(
   127→      this.scale.width / 2, this.scale.height / 2,
   128→      this.scale.width + 4, this.scale.height + 4, 0xffffff,
   129→    ).setScrollFactor(0).setDepth(DEPTH.HUD - 50).setBlendMode(Phaser.BlendModes.MULTIPLY);
   130→    this.events.once('shutdown', () => {
   131→      this.scale.off('resize', this.resizeZoom, this);
   132→      this.weapons?.destroy();
   133→      this.hud?.destroy();
   134→      this.pickups?.destroy();
   135→      audio.stopMusic();
   136→    });
   137→
   138→    audio.unlock();
   139→    audio.startMusic();
   140→    this.started = true;
   141→    this.hud.announce(`ВЫЖИВИ 15 МИНУТ, ${this.heroDef.name.toUpperCase()}!`, '#ffe9b0');

   143→      this.time.delayedCall(2100, () => this.hud.announce('⚔ КОШМАР: ВРАГИ СВИРЕПЕЕ, ОЧКИ ×1.6', '#ff8866'));

   176→  private initDifficulty(diffId?: DifficultyId): void {
   177→    let id = diffId;
   178→    if (!id && typeof window !== 'undefined') {
   179→      const saved = window.localStorage.getItem('tss-diff');
   180→      if (saved === 'normal' || saved === 'nightmare') id = saved;
   181→    }
   182→    this.diffDef = DIFFICULTIES[id ?? 'normal'];
   183→  }

   185→  private initHero(heroId?: HeroId): void {
   186→    let id = heroId;
   187→    if (!id && typeof window !== 'undefined') {
   188→      const saved = window.localStorage.getItem('tss-hero');
   189→      if (saved === 'warrior' || saved === 'archer' || saved === 'lancer' || saved === 'monk') id = saved;
   190→    }
   191→    this.heroDef = HEROES[id ?? 'warrior'];

   196→  private buildWorld(): void {
   197→    const S = WORLD.SIZE;
   198→    const m = WORLD.GRASS_MARGIN;
   199→    // Вода (фон)
   200→    const water = this.add.tileSprite(S / 2, S / 2, S, S, 'water-tile')
   201→      .setDepth(DEPTH.WATER - 20);
   202→    void water;
   203→    // Трава (арена)
   204→    const grass = this.add.tileSprite(S / 2, S / 2, S - m * 2, S - m * 2, 'grass-tile')
   205→      .setDepth(DEPTH.GROUND);
   206→    void grass;
   207→
   208→    // Анимированная пена по кромке воды
   209→    const foamStep = 192;
   210→    const foamAnim = this.anims.get('fx-foam');
   211→    const foamRing = (x: number, y: number) => {
   212→      const f = this.add.sprite(x, y, 'fx-foam').setDepth(DEPTH.WATER + 2);
   213→      f.play('fx-foam');
   214→      if (foamAnim) f.anims.setCurrentFrame(foamAnim.frames[Phaser.Math.Between(0, foamAnim.frames.length - 1)]);
   215→    };
   216→    for (let x = m; x < S - m; x += foamStep) {
   217→      foamRing(x + foamStep / 2, m);
   218→      foamRing(x + foamStep / 2, S - m);
   219→    }
   220→    for (let y = m; y < S - m; y += foamStep) {
   221→      foamRing(m, y + foamStep / 2);
   222→      foamRing(S - m, y + foamStep / 2);
   223→    }
   224→
   225→    this.physics.world.setBounds(m, m, S - m * 2, S - m * 2);
   226→
   227→    // Детерминированный декор
   228→    const rng = new Phaser.Math.RandomDataGenerator(['tiny-sword-survivors']);
   229→    const cx = S / 2, cy = S / 2;
   230→    const placed: { x: number; y: number; r: number }[] = [];
   231→    const canPlace = (x: number, y: number, r: number): boolean => {
   232→      if (Math.hypot(x - cx, y - cy) < 380) return false; // зона спавна
   233→      for (const p of placed) {
   234→        if (Math.hypot(x - p.x, y - p.y) < p.r + r) return false;
   235→      }
   236→      return true;
   237→    };
   238→    const place = (x: number, y: number, r: number) => placed.push({ x, y, r });
   239→    const randPos = (margin = 260): [number, number] => [
   240→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   241→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   242→    ];
   243→
   244→    // Замок — дом игрока
   245→    const castle = this.add.image(cx, cy - 460, 'castle').setScale(0.9)
   246→      .setDepth(DEPTH.ENTITY_BASE + cy - 460);
   247→    void castle;
   248→    place(cx, cy - 460, 200);
   249→    this.add.image(cx - 240, cy - 420, 'house2').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 420);
   250→    this.add.image(cx + 230, cy - 430, 'house1').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 430);
   251→    this.add.image(cx - 320, cy - 300, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 300);
   252→    this.add.image(cx + 320, cy - 280, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 280);
   253→    this.add.image(cx + 60, cy + 470, 'monastery').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy + 470);
   254→    place(cx + 60, cy + 470, 160);
   255→
   256→    // Дома и башни по карте
   257→    for (let i = 0; i < 8; i++) {
   258→      const [x, y] = randPos();
   259→      if (canPlace(x, y, 120)) {
   260→        const k = Phaser.Math.RND.pick(['house1', 'house2', 'house3']);
   261→        this.add.image(x, y, k).setScale(0.75).setDepth(DEPTH.ENTITY_BASE + y);
   262→        place(x, y, 120);
   263→      }
   264→    }
   265→    for (let i = 0; i < 6; i++) {
   266→      const [x, y] = randPos();
   267→      if (canPlace(x, y, 110)) {
   268→        this.add.image(x, y, 'tower').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + y);
   269→        place(x, y, 110);
   270→      }
   271→    }
   272→    // Деревья
   273→    for (let i = 0; i < 70; i++) {
   274→      const [x, y] = randPos();
   275→      if (!canPlace(x, y, 80)) continue;
   276→      const key = rng.frac() < 0.6 ? 'tree1' : 'tree3';
   277→      const sc = 0.8 + rng.frac() * 0.35;
   278→      const t = this.add.sprite(x, y, key).setScale(sc);
   279→      t.play(key === 'tree1' ? 'tree1-sway' : 'tree3-sway');
   280→      if (t.anims.currentAnim) t.setFrame(rng.between(0, 7));
   281→      t.setDepth(DEPTH.ENTITY_BASE + y);
   282→      place(x, y, 80);
   283→    }
   284→    // Камни и кусты
   285→    for (let i = 0; i < 30; i++) {
   286→      const [x, y] = randPos();
   287→      if (!canPlace(x, y, 40)) continue;
   288→      this.add.image(x, y, `rock${rng.between(1, 3)}`).setScale(0.8 + rng.frac() * 0.5)
   289→        .setDepth(DEPTH.ENTITY_BASE + y - 4);
   290→      place(x, y, 40);
   291→    }
   292→    for (let i = 0; i < 40; i++) {
   293→      const [x, y] = randPos();
   294→      if (!canPlace(x, y, 40)) continue;
   295→      const b = this.add.sprite(x, y, 'bushe1').setScale(0.7 + rng.frac() * 0.4);
   296→      b.play('bushe1-sway');
   297→      b.setFrame(rng.between(0, 7));
   298→      b.setDepth(DEPTH.ENTITY_BASE + y - 2);
   299→      place(x, y, 40);
   300→    }
   301→    // Овечки
   302→    for (let i = 0; i < 10; i++) {
   303→      const [x, y] = randPos();
   304→      if (!canPlace(x, y, 50)) continue;
   305→      const sh = this.add.sprite(x, y, 'sheep').setScale(0.8);
   306→      sh.play('sheep-idle');
   307→      sh.setDepth(DEPTH.ENTITY_BASE + y);
   308→      place(x, y, 50);
   309→    }
   310→    // Облака (дрейф вправо, относительный твин с переносом)
   311→    for (let i = 0; i < 10; i++) {
   312→      const [x, y] = [rng.between(0, S), rng.between(0, S)];
   313→      const c = this.add.image(x, y, `cloud${rng.between(1, 3)}`)
   314→        .setAlpha(0.5)
   315→        .setDepth(DEPTH.FX + 50);
   316→      this.tweens.add({
   317→        targets: c,
   318→        x: '+=700',
   319→        duration: rng.between(60, 100) * 1000,
   320→        repeat: -1,
   321→        onRepeat: () => { c.x = -320; },
   322→      });
   323→    }
   324→  }

   330→  private buildSystems(): void {
   331→    this.effects = new Effects(this);
   332→    this.pickups = new Pickups(
   333→      this,
   334→      (v) => this.addXp(v),
   335→      (v) => this.player.heal(v),
   336→      (x, y) => this.openChest(x, y),
   337→      (v, x, y) => this.addGold(v, x, y),
   338→    );
   339→    this.projectiles = new Projectiles(this, {
   340→      onExplode: (x, y, dmg, radius) => this.explode(x, y, dmg, radius, 'bomb'),
   341→    });
   342→    this.enemyGroup = this.physics.add.group();
   343→    this.enemyCtxStore = this.makeEnemyCtx();
   344→
   345→    const spawnCtx: SpawnerCtx = {
   346→      spawn: (kind, x, y) => this.spawnEnemy(kind, x, y),
   347→      spawnElite: (kind) => {
   348→        const a = Math.random() * Math.PI * 2;
   349→        this.spawnEnemy(kind, this.player.x + Math.cos(a) * 700, this.player.y + Math.sin(a) * 700);
   350→      },
   351→      spawnBoss: () => this.spawnBoss(),
   352→      playerX: () => this.player.x,
   353→      playerY: () => this.player.y,
   354→      announce: (t, c) => this.hud.announce(t, c),
   355→      spawnRadius: () => {
   356→        const cam = this.cameras.main;
   357→        return Math.hypot(cam.width, cam.height) / (2 * cam.zoom) + 80;
   358→      },
   359→    };
   360→    this.spawner = new Spawner(this, spawnCtx, this.diffDef.id);
   361→
   362→    const weaponCtx: WeaponCtx = {
   363→      player: () => this.player,
   364→      enemies: () => this.activeEnemies(),
   365→      damageEnemy: (e, dmg, kx, ky, force, src) => this.damageEnemy(e, dmg, kx, ky, force, src),
   366→      projectiles: () => this.projectiles,
   367→      effects: () => this.effects,
   368→      moveDir: () => this.lastMove,
   369→      mightMult: () => this.mightMult,
   370→      cdMult: () => this.cdMult,
   371→    };
   372→    this.weapons = new WeaponSystem(this, weaponCtx);
   373→    this.weapons.add(this.heroDef.startWeapon);
   374→    const startId = this.heroDef.startWeapon;
   375→
   376→    this.hud = new HUD(this, () => this.openPause(), {
   377→      scoreMult: this.diffDef.scoreMult,
   378→      diffName: this.diffDef.name.toUpperCase(),
   379→    });
   380→    this.hud.setIcon(startId, WEAPONS[startId].icon, 1, false);
   381→    this.ach = new AchTracker((a) => this.hud.achievementToast(a));
   382→  }

   384→  private setupCollisions(): void {
   385→    this.physics.add.overlap(this.player, this.enemyGroup, (_p, e) => {
   386→      const enemy = e as Enemy;
   387→      this.hurtPlayer(enemy.damage);
   388→    });
   389→
   390→    this.physics.add.overlap(this.projectiles.arrows, this.enemyGroup, (proj, e) => {
   391→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   392→      const enemy = e as Enemy;
   393→      if (!p.active || !enemy.active) return;
   394→      if (p.hitSet?.has(enemy)) return;
   395→      p.hitSet?.add(enemy);
   396→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 90, 'bow');
   397→      this.projectiles.consumePierce(p);
   398→    });
   399→
   400→    this.physics.add.overlap(this.projectiles.spears, this.enemyGroup, (proj, e) => {
   401→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   402→      const enemy = e as Enemy;
   403→      if (!p.active || !enemy.active) return;
   404→      if (p.hitSet?.has(enemy)) return;
   405→      p.hitSet?.add(enemy);
   406→      this.damageEnemy(enemy, p.dmg, 0, 0, 40, 'spear');
   407→      this.projectiles.consumePierce(p);
   408→    });
   409→
   410→    this.physics.add.overlap(this.projectiles.knives, this.enemyGroup, (proj, e) => {
   411→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   412→      const enemy = e as Enemy;
   413→      if (!p.active || !enemy.active) return;
   414→      if (p.hitSet?.has(enemy)) return;
   415→      p.hitSet?.add(enemy);
   416→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 70, 'knives');
   417→      this.projectiles.consumePierce(p);
   418→    });
   419→
   420→    this.physics.add.overlap(this.player, this.projectiles.enemyArrows, (_p, proj) => {
   421→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   422→      if (!p.active) return;
   423→      this.projectiles.releaseEnemyArrow(p);
   424→      this.hurtPlayer(p.dmg);
   425→    });
   426→  }

   449→  update(time: number, delta: number): void {
   450→    if (!this.started || this.gameEnded || this.victoryPending) return;
   451→
   452→    // Хитстоп: короткая заморозка мира (после убийств элит/взрывов/урона)
   453→    if (time < this.hitstopUntil) {
   454→      this.player.setVelocity(0, 0);
   455→      return;
   456→    }
   457→
   458→    const dt = Math.min(delta, 50) / 1000;
   459→
   460→    // Пауза
   461→    if (Phaser.Input.Keyboard.JustDown(this.keys.esc) || Phaser.Input.Keyboard.JustDown(this.keys.p)) {
   462→      this.openPause();
   463→      return;
   464→    }
   465→    if (Phaser.Input.Keyboard.JustDown(this.keys.m)) {
   466→      const muted = audio.toggleMute();
   467→      this.hud.announce(muted ? 'ЗВУК ВЫКЛ' : 'ЗВУК ВКЛ', '#ffe9b0');
   468→    }
   469→
   470→    this.elapsed += dt;
   471→
   472→    // Движение игрока
   473→    let vx = 0, vy = 0;
   474→    if (this.keys.a.isDown || this.keys.left.isDown) vx -= 1;
   475→    if (this.keys.d.isDown || this.keys.right.isDown) vx += 1;
   476→    if (this.keys.w.isDown || this.keys.up.isDown) vy -= 1;
   477→    if (this.keys.s.isDown || this.keys.down.isDown) vy += 1;
   478→    const jv = this.joystick.vector;
   479→    vx += jv.x; vy += jv.y;
   480→    const len = Math.hypot(vx, vy);
   481→    if (len > 1) { vx /= len; vy /= len; }
   482→    this.player.setVelocity(vx * this.player.speed, vy * this.player.speed);
   483→    this.player.setAnimForVelocity();
   484→    if (len > 0.1) this.lastMove = { x: vx / Math.max(0.001, len), y: vy / Math.max(0.001, len) };
   485→    if (Math.abs(vx) > 0.05) this.player.setFlipX(vx < 0);
   486→
   487→    // Пыль под ногами
   488→    if (len > 0.5 && Math.random() < dt * 6) {
   489→      this.effects.dust(this.player.x, this.player.y + 14, 1);
   490→    }
   491→
   492→    // Системы
   493→    this.weapons.update(delta);
   494→    this.projectiles.update(delta);
   495→    this.spawner.update(dt);
   496→
   497→    // Сундуки с сокровищами
   498→    if (!this.gameEnded) {
   499→      this.chestTimer -= dt;
   500→      if (this.chestTimer <= 0) {
   501→        this.chestTimer = 75;
   502→        this.spawnChest();
   503→      }
   504→    }
   505→    this.pickups.update(delta, this.player.x, this.player.y, this.magnetRadius, time);
   506→
   507→    // Регенерация
   508→    if (this.regen > 0 && this.player.hp < this.player.maxHp) {
   509→      this.player.heal(this.regen * dt);
   510→    }
   511→
   512→    // Сепарация врагов
   513→    this.separateEnemies();
   514→
   515→    // Деспавн далёких
   516→    if (time - this.lastDespawn > 700) {
   517→      this.lastDespawn = time;
   518→      this.despawnFar();
   519→    }
   520→
   521→    // Босс / мидбосс — приоритет полосе HP: сначала финальный босс
   522→    if (this.boss) {
   523→      if (this.boss.active) {
   524→        this.hud.updateBossBar(this.boss.hp / this.boss.maxHp);
   525→      } else {
   526→        this.boss = null;
   527→        if (this.midboss && this.midboss.active) this.hud.showBossBar(ENEMIES.midboss.name);
   528→        else this.hud.hideBossBar();
   529→      }
   530→    } else if (this.midboss) {
   531→      if (this.midboss.active) {
   532→        this.hud.updateBossBar(this.midboss.hp / this.midboss.maxHp);
   533→      } else {
   534→        this.midboss = null;
   535→        this.hud.hideBossBar();
   536→      }
   537→    }
   538→
   539→    // HUD
   540→    if (time - this.lastHudUpdate > 150) {
   541→      this.lastHudUpdate = time;
   542→      this.hud.update({
   543→        hp: this.player.hp, maxHp: this.player.maxHp,
   544→        level: this.level, xp: this.xp, xpNext: this.xpNext,
   545→        kills: this.kills, score: this.currentScore(), elapsed: this.elapsed, gold: this.gold,
   546→      });
   547→    }
   548→
   549→    // Смена освещения (троттлировано внутри)
   550→    this.updateAmbient();
   551→
   552→    // Серия убийств: окно 3с без убийств — серия сбрасывается
   553→    if (this.combo > 0) {
   554→      this.comboTimer -= dt;
   555→      if (this.comboTimer <= 0) {
   556→        this.hud.breakCombo(this.combo);
   557→        this.combo = 0;
   558→      }
   559→    }
   560→    this.hud.updateCombo(this.combo, this.comboTimer / COMBO.WINDOW);
   561→
   562→    // Индикаторы целей за экраном: сундук (золото) и крупные враги (красный)
   563→    if (time - this.lastIndicatorUpdate > 120) {
   564→      this.lastIndicatorUpdate = time;
   565→      this.updateIndicators();
   566→    }
   567→
   568→    // Левел-ап
   569→    if (this.pendingLevelUps > 0 && !this.overlayOpen && !this.gameEnded && !this.victoryPending) {
   570→      this.openLevelUp();
   571→    }
   572→  }

   594→  private damageEnemy(e: Enemy, dmg: number, kx = 0, ky = 0, force = 0, src?: WeaponId): void {
   595→    if (!e.active || e.dead) return;
   596→    let final = dmg * this.mightMult;
   597→    const crit = Math.random() < this.critChance;
   598→    if (crit) final *= 2;
   599→    final = Math.max(1, Math.round(final));
   600→    this.effects.damage(e.x, e.y - 40 * e.def.scale, final, crit);
   601→    if (Math.random() < 0.6) this.effects.hitSpark(e.x, e.y - 26 * e.def.scale);
   602→    if (force > 0) e.knockback(kx, ky, force);
   603→    this.effects.hitSound(this.time.now);
   604→    const died = e.takeDamage(final);
   605→    if (src) {
   606→      const st = this.weaponStats.get(src) ?? { dmg: 0, kills: 0 };
   607→      st.dmg += final;
   608→      if (died) st.kills++;
   609→      this.weaponStats.set(src, st);
   610→    }
   611→    if (died) {
   612→      this.onEnemyDeath(e, src);
   613→    }
   614→  }

   616→  private explode(x: number, y: number, dmg: number, radius: number, src?: WeaponId): void {

   621→      const dx = e.x - x, dy = e.y - y;
   622→      if (Math.hypot(dx, dy) <= radius + e.def.bodyRadius) {
   623→        this.damageEnemy(e, dmg, dx, dy, 320, src);
   624→      }
   625→    }
   626→  }

   641→  private onEnemyDeath(e: Enemy, src?: WeaponId): void {
   642→    if (!e.active) return;
   643→    // Убираем из мира СРАЗУ: цепные взрывы не должны повторно бить погибающего
   644→    const x = e.x, y = e.y;
   645→    this.enemyGroup.remove(e, true, true);
   646→
   647→    this.kills++;
   648→    // Серия убийств: множитель растёт с каждым убийством без паузы
   649→    this.combo++;
   650→    this.comboTimer = COMBO.WINDOW;
   651→    if (this.combo > this.bestCombo) this.bestCombo = this.combo;
   652→    const comboMult = 1 + Math.min(COMBO.BONUS_CAP - 1, this.combo * COMBO.BONUS_STEP);
   653→    this.score += Math.round(SCORE.KILL * comboMult);
   654→    if ((COMBO.MILESTONES as readonly number[]).includes(this.combo)) {
   655→      this.effects.popup(x, y - 30, `СЕРИЯ ×${this.combo}!`, '#ffd76e', 16);
   656→      if (this.combo >= 50) {
   657→        this.hud.announce(`НЕУДЕРЖИМ! СЕРИЯ ×${this.combo}`, '#ffd700');
   658→        audio.play('ach');
   659→      } else {
   660→        audio.play('levelup');
   661→      }
   662→    }
   663→    this.ach?.event({ type: 'kills', value: this.kills });
   664→    // Счётчики новых врагов → достижения
   665→    if (e.kind === 'guard') {
   666→      this.guardsSlain++;
   667→      this.ach?.event({ type: 'guards', value: this.guardsSlain });
   668→    } else if (e.kind === 'taskmaster') {
   669→      this.taskmastersSlain++;
   670→      this.ach?.event({ type: 'taskmasters', value: this.taskmastersSlain });
   671→    }
   672→    const big = e.def.isElite || e.def.isBoss || e.def.isMidboss;
   673→
   674→    // Бочка-камикадзе, убитая оружием: взрыв бьёт врагов (цепная реакция!)
   675→    if (e.def.kind === 'bomber') {
   676→      this.bomberExplosion(x, y, false, src);
   677→    }
   678→
   679→    this.effects.explosion(x, y, big ? 1.7 : e.def.scale * 1.15, big ? 2 : 1);
   680→    if (big) this.hitstop(110);
   681→    if (this.time.now - this.lastDieSound > 120) {
   682→      this.lastDieSound = this.time.now;
   683→      audio.play('die');
   684→    }
   685→    // Дроп
   686→    if (e.def.gem && e.xpValue > 0) {
   687→      const n = big ? 3 : 1;
   688→      for (let i = 0; i < n; i++) {
   689→        const ox = big ? Phaser.Math.Between(-40, 40) : 0;
   690→        const oy = big ? Phaser.Math.Between(-40, 40) : 0;
   691→        this.pickups.spawnGem(x + ox, y + oy, e.def.gem, Math.ceil(e.xpValue / n));
   692→      }
   693→    }
   694→    if (e.def.kind === 'brute' && Math.random() < 0.04) this.pickups.spawnMeat(x, y);
   695→    if (e.def.isElite) {
   696→      this.pickups.spawnMeat(x + 30, y);
   697→      this.pickups.spawnMeat(x - 30, y);
   698→      this.hud.announce('ЭЛИТА ПОВЕРЖЕНА! +25 XP', '#7CFC00');
   699→      this.elitesSlain++;
   700→      this.ach?.event({ type: 'elites', value: this.elitesSlain });
   701→    }
   702→    // Золотые кристаллы: теперь дают золото
   703→    if (Math.random() < 0.03) this.pickups.spawnGoldGem(x, y, Phaser.Math.Between(3, 8));
   704→    if (e.def.kind === 'midboss') {
   705→      this.score += 1500;
   706→      this.pickups.spawnGoldGem(x + 24, y, 40);
   707→      this.pickups.spawnGoldGem(x - 24, y, 40);
   708→      for (let i = 0; i < 3; i++) {
   709→        this.pickups.spawnGem(x + Phaser.Math.Between(-50, 50), y + Phaser.Math.Between(-50, 50), 'gem-red', 12);
   710→      }
   711→      this.hud.announce('ТЁМНЫЙ МОНАХ ПОВЕРЖЕН! +1500 ОЧКОВ', '#ffd700');
   712→    }
   713→
   714→    if (e === this.boss) {
   715→      this.victory();
   716→    }
   717→  }

   815→  private separateEnemies(): void {
   816→    this.sepGrid.clear();
   817→    const cell = 64;
   818→    const px = this.player.x, py = this.player.y;
   819→    const near: Enemy[] = [];
   820→    for (const e of this.enemyGroup.getChildren() as Enemy[]) {
   821→      if (!e.active) continue;
   822→      if (Math.abs(e.x - px) > 900 || Math.abs(e.y - py) > 700) continue;
   823→      near.push(e);
   824→      const key = (Math.floor(e.x / cell) << 16) ^ Math.floor(e.y / cell);
   825→      const arr = this.sepGrid.get(key);
   826→      if (arr) arr.push(e);
   827→      else this.sepGrid.set(key, [e]);
   828→    }
   829→    for (const e of near) {
   830→      const cx = Math.floor(e.x / cell), cy = Math.floor(e.y / cell);
   831→      for (let gx = cx - 1; gx <= cx + 1; gx++) {
   832→        for (let gy = cy - 1; gy <= cy + 1; gy++) {
   833→          const arr = this.sepGrid.get((gx << 16) ^ gy);
   834→          if (!arr) continue;
   835→          for (const o of arr) {
   836→            if (o === e || o.x < e.x) continue; // каждая пара один раз
   837→            const dx = o.x - e.x, dy = o.y - e.y;
   838→            const d = Math.hypot(dx, dy);
   839→            const min = (e.def.bodyRadius + o.def.bodyRadius) * 0.85;
   840→            if (d > 0.001 && d < min) {
   841→              const push = (min - d) / 2;
   842→              const nx = dx / d, ny = dy / d;
   843→              e.x -= nx * push; e.y -= ny * push;
   844→              o.x += nx * push; o.y += ny * push;
   845→            }
   846→          }
   847→        }
   848→      }
   849→    }
   850→  }

   856→      }
   857→    }

   862→  private addXp(v: number): void {
   863→    if (this.gameEnded || this.victoryPending) return;
   864→    this.xp += Math.max(1, Math.round(v * this.xpMult));
   865→    while (this.xp >= this.xpNext) {
   866→      this.xp -= this.xpNext;
   867→      this.level++;
   868→      this.xpNext = XP_CURVE(this.level);
   869→      this.pendingLevelUps++;
   870→      this.ach?.event({ type: 'level', value: this.level });
   871→    }
   872→  }

   926→    const w = WEAPONS[id];

   930→    if (next.count && next.count !== cur.count) parts.push(`Количество: ${next.count}`);
   931→    if (next.radius && cur.radius && next.radius !== cur.radius) parts.push(`Радиус: ${next.radius}`);
   932→    if (next.pierce && cur.pierce && next.pierce !== cur.pierce) parts.push(`Пробитие: ${next.pierce}`);

   986→  private recomputeStats(): void {
   987→    const lvl = (id: PassiveId) => this.passiveLevels.get(id) ?? 0;
   988→    const shop = this.shopLevels;
   989→    this.player.speed = this.heroDef.speed * (1 + 0.04 * shop.speed + 0.1 * lvl('boots'));
   990→    this.armorMult = 1 - 0.08 * lvl('armor');
   991→    this.magnetRadius = PLAYER_BASE.MAGNET_RADIUS * (1 + 0.2 * shop.magnet + 0.3 * lvl('magnet'));
   992→    this.regen = this.heroDef.baseRegen + 0.5 * lvl('regen');
   993→    this.mightMult = 1 + 0.06 * shop.might + 0.12 * lvl('might');
   994→    this.cdMult = Math.max(0.6, 1 - 0.08 * lvl('haste'));
   995→    this.critChance = 0.05 * shop.luck + 0.15 * lvl('luck');
   996→    this.xpMult = 1 + 0.15 * lvl('wisdom');
   997→    const newMax = this.heroDef.maxHp + 15 * shop.vitality + 25 * lvl('vitality');
   998→    if (newMax !== this.player.maxHp) {
   999→      const diff = newMax - this.player.maxHp;
  1000→      this.player.maxHp = newMax;
  1001→      this.player.hp = Math.min(newMax, this.player.hp + Math.max(0, diff));
  1002→    }

  1167→  /** Плавная смена освещения: полдень → закат → ночь (к боссу в 15:00) */
  1168→  private updateAmbient(): void {
  1169→    if (!this.ambientRect) return;
  1170→    // Троттлинг: цвет меняется медленно, хватит обновления раз в 0.5с
  1171→    if (this.lastAmbientUpdate >= 0 && this.elapsed - this.lastAmbientUpdate < 0.5) return;
  1172→    this.lastAmbientUpdate = this.elapsed;
  1173→    // Ключевые точки: [секунды, цвет]
  1174→    const keys: Array<[number, number]> = [
  1175→      [0, 0xffffff],
  1176→      [300, 0xfff3da],   // тёплый день
  1177→      [540, 0xffd9ae],   // предзакат
  1178→      [720, 0xd8b8c0],   // сумерки
  1179→      [840, 0xa8b0d8],   // ранний вечер
  1180→      [900, 0x93a0c8],   // ночь — битва с боссом
  1181→    ];
  1182→    let color = keys[keys.length - 1][1];
  1183→    for (let i = 0; i < keys.length - 1; i++) {
  1184→      const [t0, c0] = keys[i];
  1185→      const [t1, c1] = keys[i + 1];
  1186→      if (this.elapsed >= t0 && this.elapsed <= t1) {
  1187→        const f = (this.elapsed - t0) / (t1 - t0);
  1188→        const lerpCh = (a: number, b: number) => Math.round(a + (b - a) * f);
  1189→        color = Phaser.Display.Color.GetColor(
  1190→          lerpCh((c0 >> 16) & 0xff, (c1 >> 16) & 0xff),
  1191→          lerpCh((c0 >> 8) & 0xff, (c1 >> 8) & 0xff),
  1192→          lerpCh(c0 & 0xff, c1 & 0xff),
  1193→        );
  1194→        break;
  1195→      }
  1196→    }
  1197→    this.ambientRect.setFillStyle(color);
  1198→  }
