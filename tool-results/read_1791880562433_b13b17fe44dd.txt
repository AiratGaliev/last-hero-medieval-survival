   148→   148→  private buildWorld(): void {
   149→   149→    const S = WORLD.SIZE;
   150→   150→    const m = WORLD.GRASS_MARGIN;
   151→   151→    // Вода (фон)
   152→   152→    const water = this.add.tileSprite(S / 2, S / 2, S, S, 'water-tile')
   153→   153→      .setDepth(DEPTH.WATER - 20);
   154→   154→    void water;
   155→   155→    // Трава (арена)
   156→   156→    const grass = this.add.tileSprite(S / 2, S / 2, S - m * 2, S - m * 2, 'grass-tile')
   157→   157→      .setDepth(DEPTH.GROUND);
   158→   158→    void grass;
   159→   159→
   160→   160→    // Анимированная пена по кромке воды
   161→   161→    const foamStep = 192;
   162→   162→    const foamAnim = this.anims.get('fx-foam');
   163→   163→    const foamRing = (x: number, y: number) => {
   164→   164→      const f = this.add.sprite(x, y, 'fx-foam').setDepth(DEPTH.WATER + 2);
   165→   165→      f.play('fx-foam');
   166→   166→      if (foamAnim) f.anims.setCurrentFrame(foamAnim.frames[Phaser.Math.Between(0, foamAnim.frames.length - 1)]);
   167→   167→    };
   168→   168→    for (let x = m; x < S - m; x += foamStep) {
   169→   169→      foamRing(x + foamStep / 2, m);
   170→   170→      foamRing(x + foamStep / 2, S - m);
   171→   171→    }
   172→   172→    for (let y = m; y < S - m; y += foamStep) {
   173→   173→      foamRing(m, y + foamStep / 2);
   174→   174→      foamRing(S - m, y + foamStep / 2);
   175→   175→    }
   176→   176→
   177→   177→    this.physics.world.setBounds(m, m, S - m * 2, S - m * 2);
   178→   178→
   179→   179→    // Детерминированный декор
   180→   180→    const rng = new Phaser.Math.RandomDataGenerator(['tiny-sword-survivors']);
   181→   181→    const cx = S / 2, cy = S / 2;
   182→   182→    const placed: { x: number; y: number; r: number }[] = [];
   183→   183→    const canPlace = (x: number, y: number, r: number): boolean => {
   184→   184→      if (Math.hypot(x - cx, y - cy) < 380) return false; // зона спавна
   185→   185→      for (const p of placed) {
   186→   186→        if (Math.hypot(x - p.x, y - p.y) < p.r + r) return false;
   187→   187→      }
   188→   188→      return true;
   189→   189→    };
   190→   190→    const place = (x: number, y: number, r: number) => placed.push({ x, y, r });
   191→   191→    const randPos = (margin = 260): [number, number] => [
   192→   192→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   193→   193→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   194→   194→    ];
   195→   195→
   196→   196→    // Замок — дом игрока
   197→   197→    const castle = this.add.image(cx, cy - 460, 'castle').setScale(0.9)
   198→   198→      .setDepth(DEPTH.ENTITY_BASE + cy - 460);
   199→   199→    void castle;
   200→   200→    place(cx, cy - 460, 200);
   201→   201→    this.add.image(cx - 240, cy - 420, 'house2').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 420);
   202→   202→    this.add.image(cx + 230, cy - 430, 'house1').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 430);
   203→   203→    this.add.image(cx - 320, cy - 300, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 300);
   204→   204→    this.add.image(cx + 320, cy - 280, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 280);
   205→   205→    this.add.image(cx + 60, cy + 470, 'monastery').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy + 470);
   206→   206→    place(cx + 60, cy + 470, 160);
   207→   207→
   208→   208→    // Дома и башни по карте
   209→   209→    for (let i = 0; i < 8; i++) {
   210→   210→      const [x, y] = randPos();
   211→   211→      if (canPlace(x, y, 120)) {
   212→   212→        const k = Phaser.Math.RND.pick(['house1', 'house2', 'house3']);
   213→   213→        this.add.image(x, y, k).setScale(0.75).setDepth(DEPTH.ENTITY_BASE + y);
   214→   214→        place(x, y, 120);
   215→   215→      }
   216→   216→    }
   217→   217→    for (let i = 0; i < 6; i++) {
   218→   218→      const [x, y] = randPos();
   219→   219→      if (canPlace(x, y, 110)) {
   220→   220→        this.add.image(x, y, 'tower').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + y);
   221→   221→        place(x, y, 110);
   222→   222→      }
   223→   223→    }
   224→   224→    // Деревья
   225→   225→    for (let i = 0; i < 70; i++) {
   226→   226→      const [x, y] = randPos();
   227→   227→      if (!canPlace(x, y, 80)) continue;
   228→   228→      const key = rng.frac() < 0.6 ? 'tree1' : 'tree3';
   229→   229→      const sc = 0.8 + rng.frac() * 0.35;
   230→   230→      const t = this.add.sprite(x, y, key).setScale(sc);
   231→   231→      t.play(key === 'tree1' ? 'tree1-sway' : 'tree3-sway');
   232→   232→      if (t.anims.currentAnim) t.setFrame(rng.between(0, 7));
   233→   233→      t.setDepth(DEPTH.ENTITY_BASE + y);
   234→   234→      place(x, y, 80);
   235→   235→    }
   236→   236→    // Камни и кусты
   237→   237→    for (let i = 0; i < 30; i++) {
   238→   238→      const [x, y] = randPos();
   239→   239→      if (!canPlace(x, y, 40)) continue;
   240→   240→      this.add.image(x, y, `rock${rng.between(1, 3)}`).setScale(0.8 + rng.frac() * 0.5)
   241→   241→        .setDepth(DEPTH.ENTITY_BASE + y - 4);
   242→   242→      place(x, y, 40);
   243→   243→    }
   244→   244→    for (let i = 0; i < 40; i++) {
   245→   245→      const [x, y] = randPos();
   246→   246→      if (!canPlace(x, y, 40)) continue;
   247→   247→      const b = this.add.sprite(x, y, 'bushe1').setScale(0.7 + rng.frac() * 0.4);
   248→   248→      b.play('bushe1-sway');
   249→   249→      b.setFrame(rng.between(0, 7));
   250→   250→      b.setDepth(DEPTH.ENTITY_BASE + y - 2);
   251→   251→      place(x, y, 40);
   252→   252→    }
   253→   253→    // Овечки
   254→   254→    for (let i = 0; i < 10; i++) {
   255→   255→      const [x, y] = randPos();
   256→   256→      if (!canPlace(x, y, 50)) continue;
   257→   257→      const sh = this.add.sprite(x, y, 'sheep').setScale(0.8);
   258→   258→      sh.play('sheep-idle');
   259→   259→      sh.setDepth(DEPTH.ENTITY_BASE + y);
   260→   260→      place(x, y, 50);
   261→   261→    }
   262→   262→    // Облака (дрейф вправо, относительный твин с переносом)
   263→   263→    for (let i = 0; i < 10; i++) {
   264→   264→      const [x, y] = [rng.between(0, S), rng.between(0, S)];
   265→   265→      const c = this.add.image(x, y, `cloud${rng.between(1, 3)}`)
   266→   266→        .setAlpha(0.5)
   267→   267→        .setDepth(DEPTH.FX + 50);
   268→   268→      this.tweens.add({
   269→   269→        targets: c,
   270→   270→        x: '+=700',
   271→   271→        duration: rng.between(60, 100) * 1000,
   272→   272→        repeat: -1,
   273→   273→        onRepeat: () => { c.x = -320; },
   274→   274→      });
   275→   275→    }
   276→   276→  }

   335→   335→  private setupCollisions(): void {
   336→   336→    this.physics.add.overlap(this.player, this.enemyGroup, (_p, e) => {
   337→   337→      const enemy = e as Enemy;
   338→   338→      this.hurtPlayer(enemy.damage);
   339→   339→    });
   340→   340→
   341→   341→    this.physics.add.overlap(this.projectiles.arrows, this.enemyGroup, (proj, e) => {
   342→   342→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   343→   343→      const enemy = e as Enemy;
   344→   344→      if (!p.active || !enemy.active) return;
   345→   345→      if (p.hitSet?.has(enemy)) return;
   346→   346→      p.hitSet?.add(enemy);
   347→   347→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 90);
   348→   348→      this.projectiles.consumePierce(p);
   349→   349→    });
   350→   350→
   351→   351→    this.physics.add.overlap(this.projectiles.spears, this.enemyGroup, (proj, e) => {
   352→   352→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   353→   353→      const enemy = e as Enemy;
   354→   354→      if (!p.active || !enemy.active) return;
   355→   355→      if (p.hitSet?.has(enemy)) return;
   356→   356→      p.hitSet?.add(enemy);
   357→   357→      this.damageEnemy(enemy, p.dmg, 0, 0, 40);
   358→   358→      this.projectiles.consumePierce(p);
   359→   359→    });
   360→   360→
   361→   361→    this.physics.add.overlap(this.projectiles.knives, this.enemyGroup, (proj, e) => {
   362→   362→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   363→   363→      const enemy = e as Enemy;
   364→   364→      if (!p.active || !enemy.active) return;
   365→   365→      if (p.hitSet?.has(enemy)) return;
   366→   366→      p.hitSet?.add(enemy);
   367→   367→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 70);
   368→   368→      this.projectiles.consumePierce(p);
   369→   369→    });
   370→   370→
   371→   371→    this.physics.add.overlap(this.player, this.projectiles.enemyArrows, (_p, proj) => {
   372→   372→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   373→   373→      if (!p.active) return;
   374→   374→      this.projectiles.releaseEnemyArrow(p);
   375→   375→      this.hurtPlayer(p.dmg);
   376→   376→    });
   377→   377→  }

   400→   400→  update(time: number, delta: number): void {
   401→   401→    if (!this.started || this.gameEnded) return;
   402→   402→
   403→   403→    // Хитстоп: короткая заморозка мира (после убийств элит/взрывов/урона)
   404→   404→    if (time < this.hitstopUntil) {
   405→   405→      this.player.setVelocity(0, 0);
   406→   406→      return;
   407→   407→    }
   408→   408→
   409→   409→    const dt = Math.min(delta, 50) / 1000;
   410→   410→
   411→   411→    // Пауза
   412→   412→    if (Phaser.Input.Keyboard.JustDown(this.keys.esc) || Phaser.Input.Keyboard.JustDown(this.keys.p)) {
   413→   413→      this.openPause();
   414→   414→      return;
   415→   415→    }
   416→   416→    if (Phaser.Input.Keyboard.JustDown(this.keys.m)) {
   417→   417→      const muted = audio.toggleMute();
   418→   418→      this.hud.announce(muted ? 'ЗВУК ВЫКЛ' : 'ЗВУК ВКЛ', '#ffe9b0');
   419→   419→    }
   420→   420→
   421→   421→    this.elapsed += dt;
   422→   422→
   423→   423→    // Движение игрока
   424→   424→    let vx = 0, vy = 0;
   425→   425→    if (this.keys.a.isDown || this.keys.left.isDown) vx -= 1;
   426→   426→    if (this.keys.d.isDown || this.keys.right.isDown) vx += 1;
   427→   427→    if (this.keys.w.isDown || this.keys.up.isDown) vy -= 1;
   428→   428→    if (this.keys.s.isDown || this.keys.down.isDown) vy += 1;
   429→   429→    const jv = this.joystick.vector;
   430→   430→    vx += jv.x; vy += jv.y;
   431→   431→    const len = Math.hypot(vx, vy);
   432→   432→    if (len > 1) { vx /= len; vy /= len; }
   433→   433→    this.player.setVelocity(vx * this.player.speed, vy * this.player.speed);
   434→   434→    this.player.setAnimForVelocity();
   435→   435→    if (len > 0.1) this.lastMove = { x: vx / Math.max(0.001, len), y: vy / Math.max(0.001, len) };
   436→   436→    if (Math.abs(vx) > 0.05) this.player.setFlipX(vx < 0);
   437→   437→
   438→   438→    // Пыль под ногами
   439→   439→    if (len > 0.5 && Math.random() < dt * 6) {
   440→   440→      this.effects.dust(this.player.x, this.player.y + 14, 1);
   441→   441→    }
   442→   442→
   443→   443→    // Системы
   444→   444→    this.weapons.update(delta);
   445→   445→    this.projectiles.update(delta);
   446→   446→    this.spawner.update(dt);
   447→   447→
   448→   448→    // Сундуки с сокровищами
   449→   449→    if (!this.gameEnded) {
   450→   450→      this.chestTimer -= dt;
   451→   451→      if (this.chestTimer <= 0) {
   452→   452→        this.chestTimer = 75;
   453→   453→        this.spawnChest();
   454→   454→      }
   455→   455→    }
   456→   456→    this.pickups.update(delta, this.player.x, this.player.y, this.magnetRadius, time);
   457→   457→
   458→   458→    // Регенерация
   459→   459→    if (this.regen > 0 && this.player.hp < this.player.maxHp) {
   460→   460→      this.player.heal(this.regen * dt);
   461→   461→    }
   462→   462→
   463→   463→    // Сепарация врагов
   464→   464→    this.separateEnemies();
   465→   465→
   466→   466→    // Деспавн далёких
   467→   467→    if (time - this.lastDespawn > 700) {
   468→   468→      this.lastDespawn = time;
   469→   469→      this.despawnFar();
   470→   470→    }
   471→   471→
   472→   472→    // Босс
   473→   473→    if (this.boss) {
   474→   474→      if (this.boss.active) {
   475→   475→        this.hud.updateBossBar(this.boss.hp / this.boss.maxHp);
   476→   476→      } else {
   477→   477→        this.boss = null;
   478→   478→      }
   479→   479→    }
   480→   480→
   481→   481→    // HUD
   482→   482→    if (time - this.lastHudUpdate > 150) {
   483→   483→      this.lastHudUpdate = time;
   484→   484→      this.hud.update({
   485→   485→        hp: this.player.hp, maxHp: this.player.maxHp,
   486→   486→        level: this.level, xp: this.xp, xpNext: this.xpNext,
   487→   487→        kills: this.kills, score: this.currentScore(), elapsed: this.elapsed, gold: this.gold,
   488→   488→      });
   489→   489→    }
   490→   490→
   491→   491→    // Левел-ап
   492→   492→    if (this.pendingLevelUps > 0 && !this.overlayOpen && !this.gameEnded) {
   493→   493→      this.openLevelUp();
   494→   494→    }
   495→   495→  }

   556→   556→  private onEnemyDeath(e: Enemy): void {

   565→   565→    const big = e.def.isElite || e.def.isBoss;
   566→   566→

   585→   585→      }
   586→   586→    }
   587→   587→    if (e.def.kind === 'brute' && Math.random() < 0.04) this.pickups.spawnMeat(x, y);
   588→   588→    if (e.def.isElite) {
   589→   589→      this.pickups.spawnMeat(x + 30, y);
   590→   590→      this.pickups.spawnMeat(x - 30, y);

   594→   594→    }
   595→   595→    if (Math.random() < 0.03) this.pickups.spawnGem(x, y, 'gem-gold', 0);
   596→   596→
   597→   597→    if (e === this.boss) {
   598→   598→      this.victory();
   599→   599→    }
   600→   600→  }

   684→   684→  private separateEnemies(): void {
   685→   685→    this.sepGrid.clear();
   686→   686→    const cell = 64;
   687→   687→    const px = this.player.x, py = this.player.y;
   688→   688→    const near: Enemy[] = [];
   689→   689→    for (const e of this.enemyGroup.getChildren() as Enemy[]) {
   690→   690→      if (!e.active) continue;
   691→   691→      if (Math.abs(e.x - px) > 900 || Math.abs(e.y - py) > 700) continue;
   692→   692→      near.push(e);
   693→   693→      const key = (Math.floor(e.x / cell) << 16) ^ Math.floor(e.y / cell);
   694→   694→      const arr = this.sepGrid.get(key);
   695→   695→      if (arr) arr.push(e);
   696→   696→      else this.sepGrid.set(key, [e]);
   697→   697→    }
   698→   698→    for (const e of near) {
   699→   699→      const cx = Math.floor(e.x / cell), cy = Math.floor(e.y / cell);
   700→   700→      for (let gx = cx - 1; gx <= cx + 1; gx++) {
   701→   701→        for (let gy = cy - 1; gy <= cy + 1; gy++) {
   702→   702→          const arr = this.sepGrid.get((gx << 16) ^ gy);
   703→   703→          if (!arr) continue;
   704→   704→          for (const o of arr) {
   705→   705→            if (o === e || o.x < e.x) continue; // каждая пара один раз
   706→   706→            const dx = o.x - e.x, dy = o.y - e.y;
   707→   707→            const d = Math.hypot(dx, dy);
   708→   708→            const min = (e.def.bodyRadius + o.def.bodyRadius) * 0.85;
   709→   709→            if (d > 0.001 && d < min) {
   710→   710→              const push = (min - d) / 2;
   711→   711→              const nx = dx / d, ny = dy / d;
   712→   712→              e.x -= nx * push; e.y -= ny * push;
   713→   713→              o.x += nx * push; o.y += ny * push;
   714→   714→            }
   715→   715→          }
   716→   716→        }
   717→   717→      }
   718→   718→    }
   719→   719→  }

   785→   785→  private weaponUpgradeDesc(id: WeaponId, lvl: number): string {
   786→   786→    const w = WEAPONS[id];
   787→   787→    const cur = w.levels[lvl - 1];
   788→   788→    const next = w.levels[lvl];
   789→   789→    const parts: string[] = [`Урон ${cur.dmg} → ${next.dmg}`];
   790→   790→    if (next.count && next.count !== cur.count) parts.push(`Количество: ${next.count}`);
   791→   791→    if (next.radius && cur.radius && next.radius !== cur.radius) parts.push(`Радиус: ${next.radius}`);
   792→   792→    if (next.pierce && cur.pierce && next.pierce !== cur.pierce) parts.push(`Пробитие: ${next.pierce}`);
   793→   793→    return parts.join(', ');
   794→   794→  }

   815→   815→  private applyCard(card: UpgradeCard): void {
   816→   816→    switch (card.kind) {
   817→   817→      case 'weapon-new': {
   818→   818→        const id = card.id as WeaponId;
   819→   819→        this.weapons.add(id);
   820→   820→        this.hud.setIcon(id, WEAPONS[id].icon, this.weapons.level(id), false);
   821→   821→        break;
   822→   822→      }
   823→   823→      case 'weapon-up': {
   824→   824→        const id = card.id as WeaponId;
   825→   825→        this.weapons.upgrade(id);
   826→   826→        this.hud.setIcon(id, WEAPONS[id].icon, this.weapons.level(id), false);
   827→   827→        break;
   828→   828→      }
   829→   829→      case 'passive': {
   830→   830→        const id = card.id as PassiveId;
   831→   831→        const lvl = (this.passiveLevels.get(id) ?? 0) + 1;
   832→   832→        this.passiveLevels.set(id, lvl);
   833→   833→        this.hud.setIcon(id, PASSIVES[id].icon, lvl, true);
   834→   834→        this.recomputeStats();
   835→   835→        if (id === 'vitality') this.player.heal(25);
   836→   836→        break;
   837→   837→      }
   838→   838→      case 'heal': this.player.heal(40); break;
   839→   839→      case 'gold':
   840→   840→        this.gold += 100;
   841→   841→        this.ach?.event({ type: 'gold', value: this.gold });
   842→   842→        break;
   843→   843→    }
   844→   844→  }
   845→   845→
   846→   846→  private recomputeStats(): void {
   847→   847→    const lvl = (id: PassiveId) => this.passiveLevels.get(id) ?? 0;
   848→   848→    this.player.speed = this.heroDef.speed * (1 + 0.1 * lvl('boots'));
   849→   849→    this.armorMult = 1 - 0.08 * lvl('armor');
   850→   850→    this.magnetRadius = PLAYER_BASE.MAGNET_RADIUS * (1 + 0.3 * lvl('magnet'));
   851→   851→    this.regen = 0.5 * lvl('regen');
   852→   852→    this.mightMult = 1 + 0.12 * lvl('might');
   853→   853→    this.cdMult = Math.max(0.6, 1 - 0.08 * lvl('haste'));
   854→   854→    this.critChance = 0.15 * lvl('luck');
   855→   855→    this.xpMult = 1 + 0.15 * lvl('wisdom');
   856→   856→    const newMax = this.heroDef.maxHp + 25 * lvl('vitality');
   857→   857→    if (newMax !== this.player.maxHp) {
   858→   858→      const diff = newMax - this.player.maxHp;
   859→   859→      this.player.maxHp = newMax;
   860→   860→      this.player.hp = Math.min(newMax, this.player.hp + Math.max(0, diff));
   861→   861→    }
   862→   862→  }

   923→   923→      kills: this.kills,

   932→   932→    commitMeta({
   933→   933→      victory,
   934→   934→      kills: this.kills,
   935→   935→      gold: this.gold,

   943→   943→      window.dispatchEvent(new CustomEvent('tss-gameover', { detail: stats }));
   944→   944→    }
