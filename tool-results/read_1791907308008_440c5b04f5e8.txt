   276→  private createHeroCard(x: number, y: number, heroId: HeroId, cardW: number, cardH: number): void {
   277→    const hero = HEROES[heroId];
   278→    const container = this.add.container(x, y);
   279→    const paper = makePanel(this, 0, 0, 'paper', cardW, cardH);
   280→    container.add(paper);
   281→
   282→    // Живой спрайт героя
   283→    const sprite = this.add.sprite(0, -cardH * 0.27, hero.anims.idle);
   284→    sprite.setScale(hero.scale * 1.05);
   285→    sprite.play(hero.anims.idle);
   286→    container.add(sprite);
   287→
   288→    const name = this.add.text(0, cardH * 0.08, hero.name, {
   289→      fontFamily: FONT, fontSize: '17px', color: '#3b2410',
   290→    }).setOrigin(0.5);
   291→    container.add(name);
   292→
   293→    const title = this.add.text(0, cardH * 0.21, hero.title, {
   294→      fontFamily: FONT, fontSize: '11px', color: '#8a5b12',
   295→    }).setOrigin(0.5);
   296→    container.add(title);
   297→
   298→    const traits = hero.traits.map((t) => `• ${t}`).join('\n');
   299→    const body = this.add.text(0, cardH * 0.30, traits, {
   300→      fontFamily: FONT, fontSize: '11px', color: '#4a3517',
   301→      align: 'center', lineSpacing: 5, wordWrap: { width: cardW - 26 },
   302→    }).setOrigin(0.5, 0);
   303→    container.add(body);
   304→
   305→    // Иконка стартового оружия — в углу карточки (не под текстом!)
   306→    const wIcon = this.add.image(cardW / 2 - 26, -cardH / 2 + 26, WEAPONS[hero.startWeapon].icon).setDisplaySize(30, 30);
   307→    container.add(wIcon);
   308→
   309→    // Рамка выделения
   310→    const frame = this.add.graphics();
   311→    container.add(frame);
   312→
   313→    container.setSize(cardW, cardH);
   314→    container.setInteractive({ useHandCursor: true });
   315→    container.on('pointerover', () => {
   316→      if (this.selectedHero !== heroId) {
   317→        this.tweens.add({ targets: container, scale: 1.04, duration: 110 });
   318→      }
   319→    });
   320→    container.on('pointerout', () => {
   321→      if (this.selectedHero !== heroId) {
   322→        this.tweens.add({ targets: container, scale: 1, duration: 110 });
   323→      }
   324→    });
   325→    container.on('pointerdown', () => {
   326→      audio.play('click');
   327→      this.selectHero(heroId);
   328→    });
   329→
   330→    this.heroCards.push({ container, sprite, frame, heroId });
   331→  }

   333→  private createDiffChip(x: number, y: number, w: number, h: number, diffId: DifficultyId): void {
   334→    const def = DIFFICULTIES[diffId];
   335→    const container = this.add.container(x, y);
   336→    const paper = makePanel(this, 0, 0, 'paper', w, h);
   337→    container.add(paper);
   338→
   339→    const isNightmare = diffId === 'nightmare';
   340→    const icon = this.add.text(-w / 2 + 16, 0, isNightmare ? '☠' : '⚔', {
   341→      fontFamily: FONT, fontSize: '16px', color: isNightmare ? '#7a1010' : '#3b2410',
   342→    }).setOrigin(0.5);
   343→    container.add(icon);
   344→
   345→    const name = this.add.text(-w / 2 + 34, -7, def.name.toUpperCase(), {
   346→      fontFamily: FONT, fontSize: '13px', color: isNightmare ? '#7a1010' : '#3b2410',
   347→    }).setOrigin(0, 0.5);
   348→    container.add(name);
   349→
   350→    const tag = this.add.text(-w / 2 + 34, 7, def.tagline, {
   351→      fontFamily: FONT, fontSize: '8px', color: '#8a5b12',
   352→    }).setOrigin(0, 0.5);
   353→    container.add(tag);
   354→
   355→    const frame = this.add.graphics();
   356→    container.add(frame);
   357→
   358→    container.setSize(w, h);
   359→    container.setInteractive({ useHandCursor: true });
   360→    container.on('pointerover', () => {
   361→      if (this.selectedDiff !== diffId) this.tweens.add({ targets: container, scale: 1.04, duration: 100 });
   362→    });
   363→    container.on('pointerout', () => {
   364→      if (this.selectedDiff !== diffId) this.tweens.add({ targets: container, scale: 1, duration: 100 });
   365→    });
   366→    container.on('pointerdown', () => {
   367→      audio.play('click');
   368→      if (this.selectedDiff !== diffId) {
   369→        this.selectedDiff = diffId;
   370→        this.refreshSelection();
   371→      }
   372→    });
   373→
   374→    this.diffChips.push({ container, frame, diffId });
   375→  }
   376→
   377→  private selectHero(heroId: HeroId): void {
   378→    if (this.selectedHero === heroId) return;
   379→    this.selectedHero = heroId;
   380→    this.refreshSelection();
   381→  }
   382→
   383→  private refreshSelection(): void {
   384→    for (const c of this.heroCards) {
   385→      const selected = c.heroId === this.selectedHero;
   386→      c.frame.clear();
   387→      if (selected) {
   388→        c.frame.lineStyle(4, 0xd9a520, 1).strokeRect(-c.container.width / 2 + 4, -c.container.height / 2 + 4, c.container.width - 8, c.container.height - 8);
   389→        c.container.setScale(1.07);
   390→      } else {
   391→        c.container.setScale(1);
   392→      }
   393→    }
   394→    for (const d of this.diffChips) {
   395→      const selected = d.diffId === this.selectedDiff;
   396→      d.frame.clear();
   397→      d.container.setAlpha(selected ? 1 : 0.72);
   398→      if (selected) {
   399→        const w = d.container.width, h = d.container.height;
   400→        d.frame.lineStyle(3, d.diffId === 'nightmare' ? 0xc03028 : 0xd9a520, 1)
   401→          .strokeRect(-w / 2 + 3, -h / 2 + 3, w - 6, h - 6);
   402→        d.container.setScale(1.05);
   403→      } else {
   404→        d.container.setScale(1);
   405→      }
   406→    }
   407→  }
