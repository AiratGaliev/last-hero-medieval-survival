   148→  private buildWorld(): void {
   149→    const S = WORLD.SIZE;
   150→    const m = WORLD.GRASS_MARGIN;
   151→    // Вода (фон)
   152→    const water = this.add.tileSprite(S / 2, S / 2, S, S, 'water-tile')
   153→      .setDepth(DEPTH.WATER - 20);
   154→    void water;
   155→    // Трава (арена)
   156→    const grass = this.add.tileSprite(S / 2, S / 2, S - m * 2, S - m * 2, 'grass-tile')
   157→      .setDepth(DEPTH.GROUND);
   158→    void grass;
   159→
   160→    // Анимированная пена по кромке воды
   161→    const foamStep = 192;
   162→    const foamAnim = this.anims.get('fx-foam');
   163→    const foamRing = (x: number, y: number) => {
   164→      const f = this.add.sprite(x, y, 'fx-foam').setDepth(DEPTH.WATER + 2);
   165→      f.play('fx-foam');
   166→      if (foamAnim) f.anims.setCurrentFrame(foamAnim.frames[Phaser.Math.Between(0, foamAnim.frames.length - 1)]);
   167→    };
   168→    for (let x = m; x < S - m; x += foamStep) {
   169→      foamRing(x + foamStep / 2, m);
   170→      foamRing(x + foamStep / 2, S - m);
   171→    }
   172→    for (let y = m; y < S - m; y += foamStep) {
   173→      foamRing(m, y + foamStep / 2);
   174→      foamRing(S - m, y + foamStep / 2);
   175→    }
   176→
   177→    this.physics.world.setBounds(m, m, S - m * 2, S - m * 2);
   178→
   179→    // Детерминированный декор
   180→    const rng = new Phaser.Math.RandomDataGenerator(['tiny-sword-survivors']);
   181→    const cx = S / 2, cy = S / 2;
   182→    const placed: { x: number; y: number; r: number }[] = [];
   183→    const canPlace = (x: number, y: number, r: number): boolean => {
   184→      if (Math.hypot(x - cx, y - cy) < 380) return false; // зона спавна
   185→      for (const p of placed) {
   186→        if (Math.hypot(x - p.x, y - p.y) < p.r + r) return false;
   187→      }
   188→      return true;
   189→    };
   190→    const place = (x: number, y: number, r: number) => placed.push({ x, y, r });
   191→    const randPos = (margin = 260): [number, number] => [
   192→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   193→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   194→    ];
   195→
   196→    // Замок — дом игрока
   197→    const castle = this.add.image(cx, cy - 460, 'castle').setScale(0.9)
   198→      .setDepth(DEPTH.ENTITY_BASE + cy - 460);
   199→    void castle;
   200→    place(cx, cy - 460, 200);
   201→    this.add.image(cx - 240, cy - 420, 'house2').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 420);
   202→    this.add.image(cx + 230, cy - 430, 'house1').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 430);
   203→    this.add.image(cx - 320, cy - 300, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 300);
   204→    this.add.image(cx + 320, cy - 280, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 280);
   205→    this.add.image(cx + 60, cy + 470, 'monastery').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy + 470);
   206→    place(cx + 60, cy + 470, 160);
   207→
   208→    // Дома и башни по карте
   209→    for (let i = 0; i < 8; i++) {
   210→      const [x, y] = randPos();
   211→      if (canPlace(x, y, 120)) {
   212→        const k = Phaser.Math.RND.pick(['house1', 'house2', 'house3']);
   213→        this.add.image(x, y, k).setScale(0.75).setDepth(DEPTH.ENTITY_BASE + y);
   214→        place(x, y, 120);
   215→      }
   216→    }
   217→    for (let i = 0; i < 6; i++) {
   218→      const [x, y] = randPos();
   219→      if (canPlace(x, y, 110)) {
   220→        this.add.image(x, y, 'tower').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + y);
   221→        place(x, y, 110);
   222→      }
   223→    }
   224→    // Деревья
   225→    for (let i = 0; i < 70; i++) {
   226→      const [x, y] = randPos();
   227→      if (!canPlace(x, y, 80)) continue;
   228→      const key = rng.frac() < 0.6 ? 'tree1' : 'tree3';
   229→      const sc = 0.8 + rng.frac() * 0.35;
   230→      const t = this.add.sprite(x, y, key).setScale(sc);
   231→      t.play(key === 'tree1' ? 'tree1-sway' : 'tree3-sway');
   232→      if (t.anims.currentAnim) t.setFrame(rng.between(0, 7));
   233→      t.setDepth(DEPTH.ENTITY_BASE + y);
   234→      place(x, y, 80);
   235→    }
   236→    // Камни и кусты
   237→    for (let i = 0; i < 30; i++) {
   238→      const [x, y] = randPos();
   239→      if (!canPlace(x, y, 40)) continue;
   240→      this.add.image(x, y, `rock${rng.between(1, 3)}`).setScale(0.8 + rng.frac() * 0.5)
   241→        .setDepth(DEPTH.ENTITY_BASE + y - 4);
   242→      place(x, y, 40);
   243→    }
   244→    for (let i = 0; i < 40; i++) {
   245→      const [x, y] = randPos();
   246→      if (!canPlace(x, y, 40)) continue;
   247→      const b = this.add.sprite(x, y, 'bushe1').setScale(0.7 + rng.frac() * 0.4);
   248→      b.play('bushe1-sway');
   249→      b.setFrame(rng.between(0, 7));
   250→      b.setDepth(DEPTH.ENTITY_BASE + y - 2);
   251→      place(x, y, 40);
   252→    }
   253→    // Овечки
   254→    for (let i = 0; i < 10; i++) {
   255→      const [x, y] = randPos();
   256→      if (!canPlace(x, y, 50)) continue;
   257→      const sh = this.add.sprite(x, y, 'sheep').setScale(0.8);
   258→      sh.play('sheep-idle');
   259→      sh.setDepth(DEPTH.ENTITY_BASE + y);
   260→      place(x, y, 50);
   261→    }
   262→    // Облака (дрейф вправо, относительный твин с переносом)
   263→    for (let i = 0; i < 10; i++) {
   264→      const [x, y] = [rng.between(0, S), rng.between(0, S)];
   265→      const c = this.add.image(x, y, `cloud${rng.between(1, 3)}`)
   266→        .setAlpha(0.5)
   267→        .setDepth(DEPTH.FX + 50);
   268→      this.tweens.add({
   269→        targets: c,
   270→        x: '+=700',
   271→        duration: rng.between(60, 100) * 1000,
   272→        repeat: -1,
   273→        onRepeat: () => { c.x = -320; },
   274→      });
   275→    }
   276→  }

   300→        this.spawnEnemy(kind, this.player.x + Math.cos(a) * 700, this.player.y + Math.sin(a) * 700);

   318→      effects: () => this.effects,
   319→      moveDir: () => this.lastMove,
   320→      mightMult: () => this.mightMult,
   321→      cdMult: () => this.cdMult,
   322→    };

   335→  private setupCollisions(): void {
   336→    this.physics.add.overlap(this.player, this.enemyGroup, (_p, e) => {
   337→      const enemy = e as Enemy;
   338→      this.hurtPlayer(enemy.damage);
   339→    });
   340→
   341→    this.physics.add.overlap(this.projectiles.arrows, this.enemyGroup, (proj, e) => {
   342→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   343→      const enemy = e as Enemy;
   344→      if (!p.active || !enemy.active) return;
   345→      if (p.hitSet?.has(enemy)) return;
   346→      p.hitSet?.add(enemy);
   347→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 90);
   348→      this.projectiles.consumePierce(p);
   349→    });
   350→
   351→    this.physics.add.overlap(this.projectiles.spears, this.enemyGroup, (proj, e) => {
   352→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   353→      const enemy = e as Enemy;
   354→      if (!p.active || !enemy.active) return;
   355→      if (p.hitSet?.has(enemy)) return;
   356→      p.hitSet?.add(enemy);
   357→      this.damageEnemy(enemy, p.dmg, 0, 0, 40);
   358→      this.projectiles.consumePierce(p);
   359→    });
   360→
   361→    this.physics.add.overlap(this.projectiles.knives, this.enemyGroup, (proj, e) => {
   362→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   363→      const enemy = e as Enemy;
   364→      if (!p.active || !enemy.active) return;
   365→      if (p.hitSet?.has(enemy)) return;
   366→      p.hitSet?.add(enemy);
   367→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 70);
   368→      this.projectiles.consumePierce(p);
   369→    });
   370→
   371→    this.physics.add.overlap(this.player, this.projectiles.enemyArrows, (_p, proj) => {
   372→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   373→      if (!p.active) return;
   374→      this.projectiles.releaseEnemyArrow(p);
   375→      this.hurtPlayer(p.dmg);
   376→    });
   377→  }

   400→  update(time: number, delta: number): void {
   401→    if (!this.started || this.gameEnded) return;
   402→
   403→    // Хитстоп: короткая заморозка мира (после убийств элит/взрывов/урона)
   404→    if (time < this.hitstopUntil) {
   405→      this.player.setVelocity(0, 0);
   406→      return;
   407→    }
   408→
   409→    const dt = Math.min(delta, 50) / 1000;
   410→
   411→    // Пауза
   412→    if (Phaser.Input.Keyboard.JustDown(this.keys.esc) || Phaser.Input.Keyboard.JustDown(this.keys.p)) {
   413→      this.openPause();
   414→      return;
   415→    }
   416→    if (Phaser.Input.Keyboard.JustDown(this.keys.m)) {
   417→      const muted = audio.toggleMute();
   418→      this.hud.announce(muted ? 'ЗВУК ВЫКЛ' : 'ЗВУК ВКЛ', '#ffe9b0');
   419→    }
   420→
   421→    this.elapsed += dt;
   422→
   423→    // Движение игрока
   424→    let vx = 0, vy = 0;
   425→    if (this.keys.a.isDown || this.keys.left.isDown) vx -= 1;
   426→    if (this.keys.d.isDown || this.keys.right.isDown) vx += 1;
   427→    if (this.keys.w.isDown || this.keys.up.isDown) vy -= 1;
   428→    if (this.keys.s.isDown || this.keys.down.isDown) vy += 1;
   429→    const jv = this.joystick.vector;
   430→    vx += jv.x; vy += jv.y;
   431→    const len = Math.hypot(vx, vy);
   432→    if (len > 1) { vx /= len; vy /= len; }
   433→    this.player.setVelocity(vx * this.player.speed, vy * this.player.speed);
   434→    this.player.setAnimForVelocity();
   435→    if (len > 0.1) this.lastMove = { x: vx / Math.max(0.001, len), y: vy / Math.max(0.001, len) };
   436→    if (Math.abs(vx) > 0.05) this.player.setFlipX(vx < 0);
   437→
   438→    // Пыль под ногами
   439→    if (len > 0.5 && Math.random() < dt * 6) {
   440→      this.effects.dust(this.player.x, this.player.y + 14, 1);
   441→    }
   442→
   443→    // Системы
   444→    this.weapons.update(delta);
   445→    this.projectiles.update(delta);
   446→    this.spawner.update(dt);
   447→
   448→    // Сундуки с сокровищами
   449→    if (!this.gameEnded) {
   450→      this.chestTimer -= dt;
   451→      if (this.chestTimer <= 0) {
   452→        this.chestTimer = 75;
   453→        this.spawnChest();
   454→      }
   455→    }
   456→    this.pickups.update(delta, this.player.x, this.player.y, this.magnetRadius, time);
   457→
   458→    // Регенерация
   459→    if (this.regen > 0 && this.player.hp < this.player.maxHp) {
   460→      this.player.heal(this.regen * dt);
   461→    }
   462→
   463→    // Сепарация врагов
   464→    this.separateEnemies();
   465→
   466→    // Деспавн далёких
   467→    if (time - this.lastDespawn > 700) {
   468→      this.lastDespawn = time;
   469→      this.despawnFar();
   470→    }
   471→
   472→    // Босс
   473→    if (this.boss) {
   474→      if (this.boss.active) {
   475→        this.hud.updateBossBar(this.boss.hp / this.boss.maxHp);
   476→      } else {
   477→        this.boss = null;
   478→      }
   479→    }
   480→
   481→    // HUD
   482→    if (time - this.lastHudUpdate > 150) {
   483→      this.lastHudUpdate = time;
   484→      this.hud.update({
   485→        hp: this.player.hp, maxHp: this.player.maxHp,
   486→        level: this.level, xp: this.xp, xpNext: this.xpNext,
   487→        kills: this.kills, score: this.currentScore(), elapsed: this.elapsed, gold: this.gold,
   488→      });
   489→    }
   490→
   491→    // Левел-ап
   492→    if (this.pendingLevelUps > 0 && !this.overlayOpen && !this.gameEnded) {
   493→      this.openLevelUp();
   494→    }
   495→  }

   517→  private damageEnemy(e: Enemy, dmg: number, kx = 0, ky = 0, force = 0): void {
   518→    if (!e.active || e.dead) return;
   519→    let final = dmg * this.mightMult;
   520→    const crit = Math.random() < this.critChance;
   521→    if (crit) final *= 2;
   522→    final = Math.max(1, Math.round(final));
   523→    this.effects.damage(e.x, e.y - 40 * e.def.scale, final, crit);
   524→    if (force > 0) e.knockback(kx, ky, force);
   525→    this.effects.hitSound(this.time.now);
   526→    if (e.takeDamage(final)) {
   527→      this.onEnemyDeath(e);
   528→    }
   529→  }

   531→  private explode(x: number, y: number, dmg: number, radius: number): void {
   532→    this.effects.explosion(x, y, 1.1, 2);
   533→    audio.play('explosion');

   535→    for (const e of this.activeEnemies()) {
   536→      const dx = e.x - x, dy = e.y - y;
   537→      if (Math.hypot(dx, dy) <= radius + e.def.bodyRadius) {
   538→        this.damageEnemy(e, dmg, dx, dy, 320);
   539→      }
   540→    }
   541→  }

   543→  private hurtPlayer(amount: number): void {
   544→    if (this.gameEnded) return;
   545→    // Эффекты урона (звук/тряска/хитстоп) — только когда удар реально прошёл
   546→    if (this.time.now < this.player.iframesUntil) return;
   547→    const dmg = Math.max(1, Math.round(amount * this.armorMult));
   548→    const dead = this.player.damage(dmg, this.time.now);
   549→    audio.play('hurt');
   550→    this.cameras.main.shake(120, 0.0035);
   551→    this.cameras.main.flash(110, 190, 40, 30, false);
   552→    this.hitstop(80);
   553→    if (dead) this.gameOver(false);
   554→  }

   556→  private onEnemyDeath(e: Enemy): void {
   557→    if (!e.active) return;
   558→    // Убираем из мира СРАЗУ: цепные взрывы не должны повторно бить погибающего
   559→    const x = e.x, y = e.y;
   560→    this.enemyGroup.remove(e, true, true);
   561→
   562→    this.kills++;
   563→    this.score += SCORE.KILL;
   564→    this.ach?.event({ type: 'kills', value: this.kills });
   565→    const big = e.def.isElite || e.def.isBoss;
   566→
   567→    // Бочка-камикадзе, убитая оружием: взрыв бьёт врагов (цепная реакция!)
   568→    if (e.def.kind === 'bomber') {
   569→      this.bomberExplosion(x, y, false);
   570→    }
   571→
   572→    this.effects.explosion(x, y, big ? 1.7 : e.def.scale * 1.15, big ? 2 : 1);
   573→    if (big) this.hitstop(110);
   574→    if (this.time.now - this.lastDieSound > 120) {
   575→      this.lastDieSound = this.time.now;
   576→      audio.play('die');
   577→    }
   578→    // Дроп
   579→    if (e.def.gem && e.xpValue > 0) {
   580→      const n = big ? 3 : 1;
   581→      for (let i = 0; i < n; i++) {
   582→        const ox = big ? Phaser.Math.Between(-40, 40) : 0;
   583→        const oy = big ? Phaser.Math.Between(-40, 40) : 0;
   584→        this.pickups.spawnGem(x + ox, y + oy, e.def.gem, Math.ceil(e.xpValue / n));
   585→      }
   586→    }
   587→    if (e.def.kind === 'brute' && Math.random() < 0.04) this.pickups.spawnMeat(x, y);
   588→    if (e.def.isElite) {
   589→      this.pickups.spawnMeat(x + 30, y);
   590→      this.pickups.spawnMeat(x - 30, y);
   591→      this.hud.announce('ЭЛИТА ПОВЕРЖЕНА! +25 XP', '#7CFC00');
   592→      this.elitesSlain++;
   593→      this.ach?.event({ type: 'elites', value: this.elitesSlain });
   594→    }
   595→    if (Math.random() < 0.03) this.pickups.spawnGem(x, y, 'gem-gold', 0);
   596→
   597→    if (e === this.boss) {
   598→      this.victory();
   599→    }
   600→  }

   604→    const def = ENEMIES.bomber;

   617→      const e = s as Enemy;
   618→      if (!e.active || e.def.isBoss) continue;
   619→      this.damageEnemy(e, dmg * 0.8, e.x - x, e.y - y, 260);
   620→    }

   684→  private separateEnemies(): void {
   685→    this.sepGrid.clear();
   686→    const cell = 64;
   687→    const px = this.player.x, py = this.player.y;
   688→    const near: Enemy[] = [];
   689→    for (const e of this.enemyGroup.getChildren() as Enemy[]) {
   690→      if (!e.active) continue;
   691→      if (Math.abs(e.x - px) > 900 || Math.abs(e.y - py) > 700) continue;
   692→      near.push(e);
   693→      const key = (Math.floor(e.x / cell) << 16) ^ Math.floor(e.y / cell);
   694→      const arr = this.sepGrid.get(key);
   695→      if (arr) arr.push(e);
   696→      else this.sepGrid.set(key, [e]);
   697→    }
   698→    for (const e of near) {
   699→      const cx = Math.floor(e.x / cell), cy = Math.floor(e.y / cell);
   700→      for (let gx = cx - 1; gx <= cx + 1; gx++) {
   701→        for (let gy = cy - 1; gy <= cy + 1; gy++) {
   702→          const arr = this.sepGrid.get((gx << 16) ^ gy);
   703→          if (!arr) continue;
   704→          for (const o of arr) {
   705→            if (o === e || o.x < e.x) continue; // каждая пара один раз
   706→            const dx = o.x - e.x, dy = o.y - e.y;
   707→            const d = Math.hypot(dx, dy);
   708→            const min = (e.def.bodyRadius + o.def.bodyRadius) * 0.85;
   709→            if (d > 0.001 && d < min) {
   710→              const push = (min - d) / 2;
   711→              const nx = dx / d, ny = dy / d;
   712→              e.x -= nx * push; e.y -= ny * push;
   713→              o.x += nx * push; o.y += ny * push;
   714→            }
   715→          }
   716→        }
   717→      }
   718→    }
   719→  }

   737→      this.xpNext = XP_CURVE(this.level);
   738→      this.pendingLevelUps++;
   739→      this.ach?.event({ type: 'level', value: this.level });
   740→    }

   743→  private generateChoices(): UpgradeCard[] {
   744→    const pool: UpgradeCard[] = [];
   745→    for (const w of Object.values(WEAPONS)) {
   746→      const lvl = this.weapons.level(w.id);
   747→      if (lvl === 0 && this.weapons.ownedIds().length < 7) {
   748→        pool.push({
   749→          kind: 'weapon-new', id: w.id, title: w.name, subtitle: 'НОВОЕ ОРУЖИЕ',
   750→          desc: w.desc, icon: w.icon, levelFrom: 0, levelTo: 1,
   751→        });
   752→      } else if (lvl > 0 && lvl < 5) {
   753→        const next = w.levels[lvl];
   754→        pool.push({
   755→          kind: 'weapon-up', id: w.id, title: w.name, subtitle: `УРОВЕНЬ ${lvl} → ${lvl + 1}`,
   756→          desc: this.weaponUpgradeDesc(w.id, lvl), icon: w.icon, levelFrom: lvl, levelTo: lvl + 1,
   757→        });
   758→        void next;
   759→      }
   760→    }
   761→    for (const p of Object.values(PASSIVES)) {
   762→      const lvl = this.passiveLevels.get(p.id) ?? 0;
   763→      if (lvl < p.max) {
   764→        pool.push({
   765→          kind: 'passive', id: p.id, title: p.name, subtitle: lvl === 0 ? 'НОВЫЙ БОНУС' : `УРОВЕНЬ ${lvl} → ${lvl + 1}`,
   766→          desc: p.desc(lvl + 1), icon: p.icon, levelFrom: lvl, levelTo: lvl + 1,
   767→        });
   768→      }
   769→    }
   770→    Phaser.Utils.Array.Shuffle(pool);
   771→    const picked = pool.slice(0, 3);
   772→    if (picked.length === 0) {
   773→      picked.push(
   774→        { kind: 'heal', title: 'Жареное мясо', subtitle: 'ЛЕЧЕНИЕ', desc: 'Восстановить 40 HP', icon: 'icon-meat' },
   775→        { kind: 'gold', title: 'Кошель золота', subtitle: 'ЗОЛОТО', desc: '+100 золота к счёту', icon: 'icon-coin' },
   776→        { kind: 'heal', title: 'Жареное мясо', subtitle: 'ЛЕЧЕНИЕ', desc: 'Восстановить 40 HP', icon: 'icon-meat' },
   777→      );
   778→    }
   779→    while (picked.length < 3) {
   780→      picked.push({ kind: 'heal', title: 'Жареное мясо', subtitle: 'ЛЕЧЕНИЕ', desc: 'Восстановить 40 HP', icon: 'icon-meat' });
   781→    }
   782→    return picked;
   783→  }

   785→  private weaponUpgradeDesc(id: WeaponId, lvl: number): string {
   786→    const w = WEAPONS[id];
   787→    const cur = w.levels[lvl - 1];
   788→    const next = w.levels[lvl];
   789→    const parts: string[] = [`Урон ${cur.dmg} → ${next.dmg}`];
   790→    if (next.count && next.count !== cur.count) parts.push(`Количество: ${next.count}`);
   791→    if (next.radius && cur.radius && next.radius !== cur.radius) parts.push(`Радиус: ${next.radius}`);
   792→    if (next.pierce && cur.pierce && next.pierce !== cur.pierce) parts.push(`Пробитие: ${next.pierce}`);
   793→    return parts.join(', ');
   794→  }

   815→  private applyCard(card: UpgradeCard): void {
   816→    switch (card.kind) {
   817→      case 'weapon-new': {
   818→        const id = card.id as WeaponId;
   819→        this.weapons.add(id);
   820→        this.hud.setIcon(id, WEAPONS[id].icon, this.weapons.level(id), false);
   821→        break;
   822→      }
   823→      case 'weapon-up': {
   824→        const id = card.id as WeaponId;
   825→        this.weapons.upgrade(id);
   826→        this.hud.setIcon(id, WEAPONS[id].icon, this.weapons.level(id), false);
   827→        break;
   828→      }
   829→      case 'passive': {
   830→        const id = card.id as PassiveId;
   831→        const lvl = (this.passiveLevels.get(id) ?? 0) + 1;
   832→        this.passiveLevels.set(id, lvl);
   833→        this.hud.setIcon(id, PASSIVES[id].icon, lvl, true);
   834→        this.recomputeStats();
   835→        if (id === 'vitality') this.player.heal(25);
   836→        break;
   837→      }
   838→      case 'heal': this.player.heal(40); break;
   839→      case 'gold':
   840→        this.gold += 100;
   841→        this.ach?.event({ type: 'gold', value: this.gold });
   842→        break;
   843→    }
   844→  }

   846→  private recomputeStats(): void {
   847→    const lvl = (id: PassiveId) => this.passiveLevels.get(id) ?? 0;
   848→    this.player.speed = this.heroDef.speed * (1 + 0.1 * lvl('boots'));
   849→    this.armorMult = 1 - 0.08 * lvl('armor');
   850→    this.magnetRadius = PLAYER_BASE.MAGNET_RADIUS * (1 + 0.3 * lvl('magnet'));
   851→    this.regen = 0.5 * lvl('regen');
   852→    this.mightMult = 1 + 0.12 * lvl('might');
   853→    this.cdMult = Math.max(0.6, 1 - 0.08 * lvl('haste'));
   854→    this.critChance = 0.15 * lvl('luck');
   855→    this.xpMult = 1 + 0.15 * lvl('wisdom');
   856→    const newMax = this.heroDef.maxHp + 25 * lvl('vitality');
   857→    if (newMax !== this.player.maxHp) {
   858→      const diff = newMax - this.player.maxHp;
   859→      this.player.maxHp = newMax;
   860→      this.player.hp = Math.min(newMax, this.player.hp + Math.max(0, diff));
   861→    }
   862→  }

   908→    this.time.delayedCall(1900, () => this.gameOver(true));
   909→  }

   911→  private gameOver(victory: boolean): void {
   912→    if (this.gameEnded) return;
   913→    this.gameEnded = true;
   914→    this.spawner.stopped = true;
   915→    this.player.setVelocity(0, 0);
   916→    if (!victory) audio.play('gameover');
   917→    audio.stopMusic();
   918→
   919→    const finalScore = this.currentScore();
   920→    const stats = {
   921→      victory,
   922→      time: Math.floor(this.elapsed),
   923→      kills: this.kills,
   924→      level: this.level,
   925→      gold: this.gold,
   926→      score: finalScore,
   927→      hero: this.heroDef.name,
   928→      difficulty: this.diffDef.name,
   929→      newAch: (this.ach?.newlyUnlocked ?? []).map((a) => ({ name: a.name, emoji: a.emoji })),
   930→    };
   931→    // Мета-статистика (localStorage) для React-сайдбара
   932→    commitMeta({
   933→      victory,
   934→      kills: this.kills,
   935→      gold: this.gold,
   936→      score: finalScore,
   937→      playTime: Math.floor(this.elapsed),
   938→    });
   939→    this.scene.launch('GameOver', { stats });
   940→    this.scene.pause();
   941→    // Уведомляем React-слой
   942→    if (typeof window !== 'undefined') {
   943→      window.dispatchEvent(new CustomEvent('tss-gameover', { detail: stats }));
   944→    }
   945→  }

   954→  private makeEnemyCtx(): EnemyContext {
   955→    return {
   956→      playerX: () => this.player.x,
   957→      playerY: () => this.player.y,
   958→      onArcherShoot: (x, y, tx, ty) => {
   959→        const a = Math.atan2(ty - y, tx - x);
   960→        this.projectiles.enemyArrow(x, y, a, Math.round(ENEMIES.archer.damage * this.spawner.dmgMult));
   961→      },
   962→      onBossArrowRing: (x, y) => {
   963→        for (let i = 0; i < 14; i++) {
   964→          this.projectiles.enemyArrow(x, y, (i / 14) * Math.PI * 2, 14);
   965→        }
   966→      },
   967→      onMonkHeal: (x, y) => {
   968→        this.effects.heal(x, y);
   969→        for (const e of this.enemiesNear(x, y, 220)) {
   970→          if (e !== this.boss) (e as Enemy).heal(34);
   971→        }
   972→      },
   973→      onBossSummon: (x, y) => {
   974→        for (let i = 0; i < 6; i++) {
   975→          const a = (i / 6) * Math.PI * 2 + Math.random();
   976→          this.spawnEnemy('boss-minion', x + Math.cos(a) * 170, y + Math.sin(a) * 170);
   977→        }
   978→        this.hud.announce('БОСС ЗОВЁТ СВИТУ!', '#ff8866');
   979→      },
   980→      onBomberExplode: (x, y, self, selfDestruct) => {
   981→        // Уничтожаем бочку ДО взрыва: взрыв не должен бить саму бочку (иначе рекурсия)
   982→        this.enemyGroup.remove(self, true, true);
   983→        this.bomberExplosion(x, y, selfDestruct);
   984→      },
   985→      enemiesNear: (x, y, r) => this.enemiesNear(x, y, r),
   986→      healEnemy: (e, hp) => (e as Enemy).heal(hp),
   987→    };
   988→  }
