   249→  private createHeroCard(x: number, y: number, heroId: HeroId, cardW: number, cardH: number): void {
   250→    const hero = HEROES[heroId];
   251→    const container = this.add.container(x, y);
   252→    const paper = makePanel(this, 0, 0, 'paper', cardW, cardH);
   253→    container.add(paper);
   254→
   255→    // Живой спрайт героя
   256→    const sprite = this.add.sprite(0, -cardH * 0.27, hero.anims.idle);
   257→    sprite.setScale(hero.scale * 1.05);
   258→    sprite.play(hero.anims.idle);
   259→    container.add(sprite);
   260→
   261→    const name = this.add.text(0, cardH * 0.08, hero.name, {
   262→      fontFamily: FONT, fontSize: '17px', color: '#3b2410',
   263→    }).setOrigin(0.5);
   264→    container.add(name);
   265→
   266→    const title = this.add.text(0, cardH * 0.21, hero.title, {
   267→      fontFamily: FONT, fontSize: '11px', color: '#8a5b12',
   268→    }).setOrigin(0.5);
   269→    container.add(title);
   270→
   271→    const traits = hero.traits.map((t) => `• ${t}`).join('\n');
   272→    const body = this.add.text(0, cardH * 0.30, traits, {
   273→      fontFamily: FONT, fontSize: '11px', color: '#4a3517',
   274→      align: 'center', lineSpacing: 5, wordWrap: { width: cardW - 26 },
   275→    }).setOrigin(0.5, 0);
   276→    container.add(body);
   277→
   278→    // Иконка стартового оружия — в углу карточки (не под текстом!)
   279→    const wIcon = this.add.image(cardW / 2 - 26, -cardH / 2 + 26, WEAPONS[hero.startWeapon].icon).setDisplaySize(30, 30);
   280→    container.add(wIcon);
   281→
   282→    // Рамка выделения
   283→    const frame = this.add.graphics();
   284→    container.add(frame);
   285→
   286→    container.setSize(cardW, cardH);
   287→    container.setInteractive({ useHandCursor: true });
   288→    container.on('pointerover', () => {
   289→      if (this.selectedHero !== heroId) {
   290→        this.tweens.add({ targets: container, scale: 1.04, duration: 110 });
   291→      }
   292→    });
   293→    container.on('pointerout', () => {
   294→      if (this.selectedHero !== heroId) {
   295→        this.tweens.add({ targets: container, scale: 1, duration: 110 });
   296→      }
   297→    });
   298→    container.on('pointerdown', () => {
   299→      audio.play('click');
   300→      this.selectHero(heroId);
   301→    });
   302→
   303→    this.heroCards.push({ container, sprite, frame, heroId });
   304→  }

   306→  private createDiffChip(x: number, y: number, w: number, h: number, diffId: DifficultyId): void {
   307→    const def = DIFFICULTIES[diffId];
   308→    const container = this.add.container(x, y);
   309→    const paper = makePanel(this, 0, 0, 'paper', w, h);
   310→    container.add(paper);
   311→
   312→    const isNightmare = diffId === 'nightmare';
   313→    const icon = this.add.text(-w / 2 + 16, 0, isNightmare ? '☠' : '⚔', {
   314→      fontFamily: FONT, fontSize: '16px', color: isNightmare ? '#7a1010' : '#3b2410',
   315→    }).setOrigin(0.5);
   316→    container.add(icon);
   317→
   318→    const name = this.add.text(-w / 2 + 34, -7, def.name.toUpperCase(), {
   319→      fontFamily: FONT, fontSize: '13px', color: isNightmare ? '#7a1010' : '#3b2410',
   320→    }).setOrigin(0, 0.5);
   321→    container.add(name);
   322→
   323→    const tag = this.add.text(-w / 2 + 34, 7, def.tagline, {
   324→      fontFamily: FONT, fontSize: '8px', color: '#8a5b12',
   325→    }).setOrigin(0, 0.5);
   326→    container.add(tag);
   327→
   328→    const frame = this.add.graphics();
   329→    container.add(frame);
   330→
   331→    container.setSize(w, h);
   332→    container.setInteractive({ useHandCursor: true });
   333→    container.on('pointerover', () => {
   334→      if (this.selectedDiff !== diffId) this.tweens.add({ targets: container, scale: 1.04, duration: 100 });
   335→    });
   336→    container.on('pointerout', () => {
   337→      if (this.selectedDiff !== diffId) this.tweens.add({ targets: container, scale: 1, duration: 100 });
   338→    });
   339→    container.on('pointerdown', () => {
   340→      audio.play('click');
   341→      if (this.selectedDiff !== diffId) {
   342→        this.selectedDiff = diffId;
   343→        this.refreshSelection();
   344→      }
   345→    });
   346→
   347→    this.diffChips.push({ container, frame, diffId });
   348→  }
   349→
   350→  private selectHero(heroId: HeroId): void {
   351→    if (this.selectedHero === heroId) return;
   352→    this.selectedHero = heroId;
   353→    this.refreshSelection();
   354→  }
   355→
   356→  private refreshSelection(): void {
   357→    for (const c of this.heroCards) {
   358→      const selected = c.heroId === this.selectedHero;
   359→      c.frame.clear();
   360→      if (selected) {
   361→        c.frame.lineStyle(4, 0xd9a520, 1).strokeRect(-c.container.width / 2 + 4, -c.container.height / 2 + 4, c.container.width - 8, c.container.height - 8);
   362→        c.container.setScale(1.07);
   363→      } else {
   364→        c.container.setScale(1);
   365→      }
   366→    }
   367→    for (const d of this.diffChips) {
   368→      const selected = d.diffId === this.selectedDiff;
   369→      d.frame.clear();
   370→      d.container.setAlpha(selected ? 1 : 0.72);
   371→      if (selected) {
   372→        const w = d.container.width, h = d.container.height;
   373→        d.frame.lineStyle(3, d.diffId === 'nightmare' ? 0xc03028 : 0xd9a520, 1)
   374→          .strokeRect(-w / 2 + 3, -h / 2 + 3, w - 6, h - 6);
   375→        d.container.setScale(1.05);
   376→      } else {
   377→        d.container.setScale(1);
   378→      }
   379→    }
   380→  }
