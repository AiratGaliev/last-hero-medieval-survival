   159→  private buildWorld(): void {
   160→    const S = WORLD.SIZE;
   161→    const m = WORLD.GRASS_MARGIN;
   162→    // Вода (фон)
   163→    const water = this.add.tileSprite(S / 2, S / 2, S, S, 'water-tile')
   164→      .setDepth(DEPTH.WATER - 20);
   165→    void water;
   166→    // Трава (арена)
   167→    const grass = this.add.tileSprite(S / 2, S / 2, S - m * 2, S - m * 2, 'grass-tile')
   168→      .setDepth(DEPTH.GROUND);
   169→    void grass;
   170→
   171→    // Анимированная пена по кромке воды
   172→    const foamStep = 192;
   173→    const foamAnim = this.anims.get('fx-foam');
   174→    const foamRing = (x: number, y: number) => {
   175→      const f = this.add.sprite(x, y, 'fx-foam').setDepth(DEPTH.WATER + 2);
   176→      f.play('fx-foam');
   177→      if (foamAnim) f.anims.setCurrentFrame(foamAnim.frames[Phaser.Math.Between(0, foamAnim.frames.length - 1)]);
   178→    };
   179→    for (let x = m; x < S - m; x += foamStep) {
   180→      foamRing(x + foamStep / 2, m);
   181→      foamRing(x + foamStep / 2, S - m);
   182→    }
   183→    for (let y = m; y < S - m; y += foamStep) {
   184→      foamRing(m, y + foamStep / 2);
   185→      foamRing(S - m, y + foamStep / 2);
   186→    }
   187→
   188→    this.physics.world.setBounds(m, m, S - m * 2, S - m * 2);
   189→
   190→    // Детерминированный декор
   191→    const rng = new Phaser.Math.RandomDataGenerator(['tiny-sword-survivors']);
   192→    const cx = S / 2, cy = S / 2;
   193→    const placed: { x: number; y: number; r: number }[] = [];
   194→    const canPlace = (x: number, y: number, r: number): boolean => {
   195→      if (Math.hypot(x - cx, y - cy) < 380) return false; // зона спавна
   196→      for (const p of placed) {
   197→        if (Math.hypot(x - p.x, y - p.y) < p.r + r) return false;
   198→      }
   199→      return true;
   200→    };
   201→    const place = (x: number, y: number, r: number) => placed.push({ x, y, r });
   202→    const randPos = (margin = 260): [number, number] => [
   203→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   204→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   205→    ];
   206→
   207→    // Замок — дом игрока
   208→    const castle = this.add.image(cx, cy - 460, 'castle').setScale(0.9)
   209→      .setDepth(DEPTH.ENTITY_BASE + cy - 460);
   210→    void castle;
   211→    place(cx, cy - 460, 200);
   212→    this.add.image(cx - 240, cy - 420, 'house2').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 420);
   213→    this.add.image(cx + 230, cy - 430, 'house1').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 430);
   214→    this.add.image(cx - 320, cy - 300, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 300);
   215→    this.add.image(cx + 320, cy - 280, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 280);
   216→    this.add.image(cx + 60, cy + 470, 'monastery').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy + 470);
   217→    place(cx + 60, cy + 470, 160);
   218→
   219→    // Дома и башни по карте
   220→    for (let i = 0; i < 8; i++) {
   221→      const [x, y] = randPos();
   222→      if (canPlace(x, y, 120)) {
   223→        const k = Phaser.Math.RND.pick(['house1', 'house2', 'house3']);
   224→        this.add.image(x, y, k).setScale(0.75).setDepth(DEPTH.ENTITY_BASE + y);
   225→        place(x, y, 120);
   226→      }
   227→    }
   228→    for (let i = 0; i < 6; i++) {
   229→      const [x, y] = randPos();
   230→      if (canPlace(x, y, 110)) {
   231→        this.add.image(x, y, 'tower').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + y);
   232→        place(x, y, 110);
   233→      }
   234→    }
   235→    // Деревья
   236→    for (let i = 0; i < 70; i++) {
   237→      const [x, y] = randPos();
   238→      if (!canPlace(x, y, 80)) continue;
   239→      const key = rng.frac() < 0.6 ? 'tree1' : 'tree3';
   240→      const sc = 0.8 + rng.frac() * 0.35;
   241→      const t = this.add.sprite(x, y, key).setScale(sc);
   242→      t.play(key === 'tree1' ? 'tree1-sway' : 'tree3-sway');
   243→      if (t.anims.currentAnim) t.setFrame(rng.between(0, 7));
   244→      t.setDepth(DEPTH.ENTITY_BASE + y);
   245→      place(x, y, 80);
   246→    }
   247→    // Камни и кусты
   248→    for (let i = 0; i < 30; i++) {
   249→      const [x, y] = randPos();
   250→      if (!canPlace(x, y, 40)) continue;
   251→      this.add.image(x, y, `rock${rng.between(1, 3)}`).setScale(0.8 + rng.frac() * 0.5)
   252→        .setDepth(DEPTH.ENTITY_BASE + y - 4);
   253→      place(x, y, 40);
   254→    }
   255→    for (let i = 0; i < 40; i++) {
   256→      const [x, y] = randPos();
   257→      if (!canPlace(x, y, 40)) continue;
   258→      const b = this.add.sprite(x, y, 'bushe1').setScale(0.7 + rng.frac() * 0.4);
   259→      b.play('bushe1-sway');
   260→      b.setFrame(rng.between(0, 7));
   261→      b.setDepth(DEPTH.ENTITY_BASE + y - 2);
   262→      place(x, y, 40);
   263→    }
   264→    // Овечки
   265→    for (let i = 0; i < 10; i++) {
   266→      const [x, y] = randPos();
   267→      if (!canPlace(x, y, 50)) continue;
   268→      const sh = this.add.sprite(x, y, 'sheep').setScale(0.8);
   269→      sh.play('sheep-idle');
   270→      sh.setDepth(DEPTH.ENTITY_BASE + y);
   271→      place(x, y, 50);
   272→    }
   273→    // Облака (дрейф вправо, относительный твин с переносом)
   274→    for (let i = 0; i < 10; i++) {
   275→      const [x, y] = [rng.between(0, S), rng.between(0, S)];
   276→      const c = this.add.image(x, y, `cloud${rng.between(1, 3)}`)
   277→        .setAlpha(0.5)
   278→        .setDepth(DEPTH.FX + 50);
   279→      this.tweens.add({
   280→        targets: c,
   281→        x: '+=700',
   282→        duration: rng.between(60, 100) * 1000,
   283→        repeat: -1,
   284→        onRepeat: () => { c.x = -320; },
   285→      });
   286→    }
   287→  }

   347→  private setupCollisions(): void {
   348→    this.physics.add.overlap(this.player, this.enemyGroup, (_p, e) => {
   349→      const enemy = e as Enemy;
   350→      this.hurtPlayer(enemy.damage);
   351→    });
   352→
   353→    this.physics.add.overlap(this.projectiles.arrows, this.enemyGroup, (proj, e) => {
   354→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   355→      const enemy = e as Enemy;
   356→      if (!p.active || !enemy.active) return;
   357→      if (p.hitSet?.has(enemy)) return;
   358→      p.hitSet?.add(enemy);
   359→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 90);
   360→      this.projectiles.consumePierce(p);
   361→    });
   362→
   363→    this.physics.add.overlap(this.projectiles.spears, this.enemyGroup, (proj, e) => {
   364→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   365→      const enemy = e as Enemy;
   366→      if (!p.active || !enemy.active) return;
   367→      if (p.hitSet?.has(enemy)) return;
   368→      p.hitSet?.add(enemy);
   369→      this.damageEnemy(enemy, p.dmg, 0, 0, 40);
   370→      this.projectiles.consumePierce(p);
   371→    });
   372→
   373→    this.physics.add.overlap(this.projectiles.knives, this.enemyGroup, (proj, e) => {
   374→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   375→      const enemy = e as Enemy;
   376→      if (!p.active || !enemy.active) return;
   377→      if (p.hitSet?.has(enemy)) return;
   378→      p.hitSet?.add(enemy);
   379→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 70);
   380→      this.projectiles.consumePierce(p);
   381→    });
   382→
   383→    this.physics.add.overlap(this.player, this.projectiles.enemyArrows, (_p, proj) => {
   384→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   385→      if (!p.active) return;
   386→      this.projectiles.releaseEnemyArrow(p);
   387→      this.hurtPlayer(p.dmg);
   388→    });
   389→  }

   411→
   412→  update(time: number, delta: number): void {
   413→    if (!this.started || this.gameEnded || this.victoryPending) return;
   414→
   415→    // Хитстоп: короткая заморозка мира (после убийств элит/взрывов/урона)
   416→    if (time < this.hitstopUntil) {
   417→      this.player.setVelocity(0, 0);
   418→      return;
   419→    }
   420→
   421→    const dt = Math.min(delta, 50) / 1000;
   422→
   423→    // Пауза
   424→    if (Phaser.Input.Keyboard.JustDown(this.keys.esc) || Phaser.Input.Keyboard.JustDown(this.keys.p)) {
   425→      this.openPause();
   426→      return;
   427→    }
   428→    if (Phaser.Input.Keyboard.JustDown(this.keys.m)) {
   429→      const muted = audio.toggleMute();
   430→      this.hud.announce(muted ? 'ЗВУК ВЫКЛ' : 'ЗВУК ВКЛ', '#ffe9b0');
   431→    }
   432→
   433→    this.elapsed += dt;
   434→
   435→    // Движение игрока
   436→    let vx = 0, vy = 0;
   437→    if (this.keys.a.isDown || this.keys.left.isDown) vx -= 1;
   438→    if (this.keys.d.isDown || this.keys.right.isDown) vx += 1;
   439→    if (this.keys.w.isDown || this.keys.up.isDown) vy -= 1;
   440→    if (this.keys.s.isDown || this.keys.down.isDown) vy += 1;
   441→    const jv = this.joystick.vector;
   442→    vx += jv.x; vy += jv.y;
   443→    const len = Math.hypot(vx, vy);
   444→    if (len > 1) { vx /= len; vy /= len; }
   445→    this.player.setVelocity(vx * this.player.speed, vy * this.player.speed);
   446→    this.player.setAnimForVelocity();
   447→    if (len > 0.1) this.lastMove = { x: vx / Math.max(0.001, len), y: vy / Math.max(0.001, len) };
   448→    if (Math.abs(vx) > 0.05) this.player.setFlipX(vx < 0);
   449→
   450→    // Пыль под ногами
   451→    if (len > 0.5 && Math.random() < dt * 6) {
   452→      this.effects.dust(this.player.x, this.player.y + 14, 1);
   453→    }
   454→
   455→    // Системы
   456→    this.weapons.update(delta);
   457→    this.projectiles.update(delta);
   458→    this.spawner.update(dt);
   459→
   460→    // Сундуки с сокровищами
   461→    if (!this.gameEnded) {
   462→      this.chestTimer -= dt;
   463→      if (this.chestTimer <= 0) {
   464→        this.chestTimer = 75;
   465→        this.spawnChest();
   466→      }
   467→    }
   468→    this.pickups.update(delta, this.player.x, this.player.y, this.magnetRadius, time);
   469→
   470→    // Регенерация
   471→    if (this.regen > 0 && this.player.hp < this.player.maxHp) {
   472→      this.player.heal(this.regen * dt);
   473→    }
   474→
   475→    // Сепарация врагов
   476→    this.separateEnemies();
   477→
   478→    // Деспавн далёких
   479→    if (time - this.lastDespawn > 700) {
   480→      this.lastDespawn = time;
   481→      this.despawnFar();
   482→    }
   483→
   484→    // Босс / мидбосс — приоритет полосе HP: сначала финальный босс
   485→    if (this.boss) {
   486→      if (this.boss.active) {
   487→        this.hud.updateBossBar(this.boss.hp / this.boss.maxHp);
   488→      } else {
   489→        this.boss = null;
   490→        if (this.midboss && this.midboss.active) this.hud.showBossBar(ENEMIES.midboss.name);
   491→        else this.hud.hideBossBar();
   492→      }
   493→    } else if (this.midboss) {
   494→      if (this.midboss.active) {
   495→        this.hud.updateBossBar(this.midboss.hp / this.midboss.maxHp);
   496→      } else {
   497→        this.midboss = null;
   498→        this.hud.hideBossBar();
   499→      }
   500→    }
   501→
   502→    // HUD
   503→    if (time - this.lastHudUpdate > 150) {
   504→      this.lastHudUpdate = time;
   505→      this.hud.update({
   506→        hp: this.player.hp, maxHp: this.player.maxHp,
   507→        level: this.level, xp: this.xp, xpNext: this.xpNext,
   508→        kills: this.kills, score: this.currentScore(), elapsed: this.elapsed, gold: this.gold,
   509→      });
   510→    }
   511→
   512→    // Левел-ап
   513→    if (this.pendingLevelUps > 0 && !this.overlayOpen && !this.gameEnded && !this.victoryPending) {
   514→      this.openLevelUp();
   515→    }
   516→  }

   538→  private damageEnemy(e: Enemy, dmg: number, kx = 0, ky = 0, force = 0): void {
   539→    if (!e.active || e.dead) return;
   540→    let final = dmg * this.mightMult;
   541→    const crit = Math.random() < this.critChance;
   542→    if (crit) final *= 2;
   543→    final = Math.max(1, Math.round(final));
   544→    this.effects.damage(e.x, e.y - 40 * e.def.scale, final, crit);
   545→    if (Math.random() < 0.6) this.effects.hitSpark(e.x, e.y - 26 * e.def.scale);
   546→    if (force > 0) e.knockback(kx, ky, force);
   547→    this.effects.hitSound(this.time.now);
   548→    if (e.takeDamage(final)) {
   549→      this.onEnemyDeath(e);
   550→    }

   578→  private onEnemyDeath(e: Enemy): void {
   579→    if (!e.active) return;
   580→    // Убираем из мира СРАЗУ: цепные взрывы не должны повторно бить погибающего
   581→    const x = e.x, y = e.y;
   582→    this.enemyGroup.remove(e, true, true);
   583→
   584→    this.kills++;
   585→    this.score += SCORE.KILL;
   586→    this.ach?.event({ type: 'kills', value: this.kills });
   587→    const big = e.def.isElite || e.def.isBoss || e.def.isMidboss;
   588→
   589→    // Бочка-камикадзе, убитая оружием: взрыв бьёт врагов (цепная реакция!)
   590→    if (e.def.kind === 'bomber') {
   591→      this.bomberExplosion(x, y, false);
   592→    }
   593→
   594→    this.effects.explosion(x, y, big ? 1.7 : e.def.scale * 1.15, big ? 2 : 1);
   595→    if (big) this.hitstop(110);
   596→    if (this.time.now - this.lastDieSound > 120) {
   597→      this.lastDieSound = this.time.now;
   598→      audio.play('die');
   599→    }
   600→    // Дроп
   601→    if (e.def.gem && e.xpValue > 0) {
   602→      const n = big ? 3 : 1;
   603→      for (let i = 0; i < n; i++) {
   604→        const ox = big ? Phaser.Math.Between(-40, 40) : 0;
   605→        const oy = big ? Phaser.Math.Between(-40, 40) : 0;
   606→        this.pickups.spawnGem(x + ox, y + oy, e.def.gem, Math.ceil(e.xpValue / n));
   607→      }
   608→    }
   609→    if (e.def.kind === 'brute' && Math.random() < 0.04) this.pickups.spawnMeat(x, y);
   610→    if (e.def.isElite) {
   611→      this.pickups.spawnMeat(x + 30, y);
   612→      this.pickups.spawnMeat(x - 30, y);
   613→      this.hud.announce('ЭЛИТА ПОВЕРЖЕНА! +25 XP', '#7CFC00');
   614→      this.elitesSlain++;
   615→      this.ach?.event({ type: 'elites', value: this.elitesSlain });
   616→    }
   617→    // Золотые кристаллы: теперь дают золото
   618→    if (Math.random() < 0.03) this.pickups.spawnGoldGem(x, y, Phaser.Math.Between(3, 8));
   619→    if (e.def.kind === 'midboss') {
   620→      this.score += 1500;
   621→      this.pickups.spawnGoldGem(x + 24, y, 40);
   622→      this.pickups.spawnGoldGem(x - 24, y, 40);
   623→      for (let i = 0; i < 3; i++) {
   624→        this.pickups.spawnGem(x + Phaser.Math.Between(-50, 50), y + Phaser.Math.Between(-50, 50), 'gem-red', 12);
   625→      }
   626→      this.hud.announce('ТЁМНЫЙ МОНАХ ПОВЕРЖЕН! +1500 ОЧКОВ', '#ffd700');
   627→    }
   628→
   629→    if (e === this.boss) {
   630→      this.victory();
   631→    }
   632→  }

   723→  private separateEnemies(): void {
   724→    this.sepGrid.clear();
   725→    const cell = 64;
   726→    const px = this.player.x, py = this.player.y;
   727→    const near: Enemy[] = [];
   728→    for (const e of this.enemyGroup.getChildren() as Enemy[]) {
   729→      if (!e.active) continue;
   730→      if (Math.abs(e.x - px) > 900 || Math.abs(e.y - py) > 700) continue;
   731→      near.push(e);
   732→      const key = (Math.floor(e.x / cell) << 16) ^ Math.floor(e.y / cell);
   733→      const arr = this.sepGrid.get(key);
   734→      if (arr) arr.push(e);
   735→      else this.sepGrid.set(key, [e]);
   736→    }
   737→    for (const e of near) {
   738→      const cx = Math.floor(e.x / cell), cy = Math.floor(e.y / cell);
   739→      for (let gx = cx - 1; gx <= cx + 1; gx++) {
   740→        for (let gy = cy - 1; gy <= cy + 1; gy++) {
   741→          const arr = this.sepGrid.get((gx << 16) ^ gy);
   742→          if (!arr) continue;
   743→          for (const o of arr) {
   744→            if (o === e || o.x < e.x) continue; // каждая пара один раз
   745→            const dx = o.x - e.x, dy = o.y - e.y;
   746→            const d = Math.hypot(dx, dy);
   747→            const min = (e.def.bodyRadius + o.def.bodyRadius) * 0.85;
   748→            if (d > 0.001 && d < min) {
   749→              const push = (min - d) / 2;
   750→              const nx = dx / d, ny = dy / d;
   751→              e.x -= nx * push; e.y -= ny * push;
   752→              o.x += nx * push; o.y += ny * push;
   753→            }
   754→          }
   755→        }
   756→      }
   757→    }
   758→  }

   833→  private weaponUpgradeDesc(id: WeaponId, lvl: number): string {
   834→    const w = WEAPONS[id];
   835→    const cur = w.levels[lvl - 1];
   836→    const next = w.levels[lvl];
   837→    const parts: string[] = [`Урон ${cur.dmg} → ${next.dmg}`];
   838→    if (next.count && next.count !== cur.count) parts.push(`Количество: ${next.count}`);
   839→    if (next.radius && cur.radius && next.radius !== cur.radius) parts.push(`Радиус: ${next.radius}`);
   840→    if (next.pierce && cur.pierce && next.pierce !== cur.pierce) parts.push(`Пробитие: ${next.pierce}`);
   841→    return parts.join(', ');

   863→  private applyCard(card: UpgradeCard): void {
   864→    switch (card.kind) {
   865→      case 'weapon-new': {
   866→        const id = card.id as WeaponId;
   867→        this.weapons.add(id);
   868→        this.hud.setIcon(id, WEAPONS[id].icon, this.weapons.level(id), false);

   870→      }
   871→      case 'weapon-up': {
   872→        const id = card.id as WeaponId;
   873→        this.weapons.upgrade(id);
   874→        this.hud.setIcon(id, WEAPONS[id].icon, this.weapons.level(id), false);
   875→        break;
   876→      }
   877→      case 'passive': {
   878→        const id = card.id as PassiveId;
   879→        const lvl = (this.passiveLevels.get(id) ?? 0) + 1;
   880→        this.passiveLevels.set(id, lvl);
   881→        this.hud.setIcon(id, PASSIVES[id].icon, lvl, true);
   882→        this.recomputeStats();
   883→        if (id === 'vitality') this.player.heal(25);
   884→        break;
   885→      }
   886→      case 'heal': this.player.heal(40); break;
   887→      case 'gold':
   888→        this.gold += 100;
   889→        this.ach?.event({ type: 'gold', value: this.gold });
   890→        break;
   891→    }
   892→  }
   893→
   894→  private recomputeStats(): void {
   895→    const lvl = (id: PassiveId) => this.passiveLevels.get(id) ?? 0;
   896→    const shop = this.shopLevels;
   897→    this.player.speed = this.heroDef.speed * (1 + 0.04 * shop.speed + 0.1 * lvl('boots'));
   898→    this.armorMult = 1 - 0.08 * lvl('armor');
   899→    this.magnetRadius = PLAYER_BASE.MAGNET_RADIUS * (1 + 0.2 * shop.magnet + 0.3 * lvl('magnet'));
   900→    this.regen = this.heroDef.baseRegen + 0.5 * lvl('regen');
   901→    this.mightMult = 1 + 0.06 * shop.might + 0.12 * lvl('might');
   902→    this.cdMult = Math.max(0.6, 1 - 0.08 * lvl('haste'));
   903→    this.critChance = 0.05 * shop.luck + 0.15 * lvl('luck');
   904→    this.xpMult = 1 + 0.15 * lvl('wisdom');
   905→    const newMax = this.heroDef.maxHp + 15 * shop.vitality + 25 * lvl('vitality');
   906→    if (newMax !== this.player.maxHp) {
   907→      const diff = newMax - this.player.maxHp;
   908→      this.player.maxHp = newMax;
   909→      this.player.hp = Math.min(newMax, this.player.hp + Math.max(0, diff));
   910→    }
   911→  }

   929→  private currentScore(): number {
   930→    const base = this.score
   931→      + Math.floor(this.elapsed / 60) * SCORE.MINUTE
   932→      + this.level * SCORE.LEVEL
   933→      + this.gold;
   934→    return Math.round(base * this.diffDef.scoreMult);
   935→  }

   937→  private victory(): void {
   938→    if (this.gameEnded || this.victoryPending) return;
   939→    this.victoryPending = true;
   940→    this.spawner.stopped = true;
   941→    this.player.setVelocity(0, 0);
   942→    this.score += SCORE.BOSS;
   943→    this.hud.hideBossBar();
   944→    this.hud.announce('ПОБЕДА!!! БОСС ПОВЕРЖЕН!', '#ffd700');
   945→    audio.play('victory');
   946→    this.cameras.main.shake(500, 0.006);
   947→    this.hitstop(200);
   948→    this.ach?.event({ type: 'victory', difficulty: this.diffDef.id });
   949→    // Финальный фейерверк
   950→    for (let i = 0; i < 8; i++) {
   951→      this.time.delayedCall(i * 160, () => {
   952→        this.effects.explosion(
   953→          this.player.x + Phaser.Math.Between(-300, 300),
   954→          this.player.y + Phaser.Math.Between(-300, 300), 1.6, 2,
   955→        );
   956→      });
   957→    }
   958→    // Собрать все кристаллы
   959→    this.pickups.vacuumAll(this.player.x, this.player.y);
   960→    this.time.delayedCall(1900, () => this.gameOver(true));
   961→  }

   963→  private gameOver(victory: boolean): void {
   964→    if (this.gameEnded) return;
   965→    this.gameEnded = true;
   966→    this.spawner.stopped = true;
   967→    this.player.setVelocity(0, 0);
   968→    if (!victory) audio.play('gameover');
   969→    audio.stopMusic();
   970→
   971→    const finalScore = this.currentScore();
   972→    const stats = {
   973→      victory,
   974→      time: Math.floor(this.elapsed),
   975→      kills: this.kills,
   976→      level: this.level,
   977→      gold: this.gold,
   978→      score: finalScore,
   979→      hero: this.heroDef.name,
   980→      difficulty: this.diffDef.name,
   981→      newAch: (this.ach?.newlyUnlocked ?? []).map((a) => ({ name: a.name, emoji: a.emoji })),
   982→    };
   983→    // Мета-статистика (localStorage) для React-сайдбара
   984→    commitMeta({
   985→      victory,
   986→      kills: this.kills,
   987→      gold: this.gold,
   988→      score: finalScore,
   989→      playTime: Math.floor(this.elapsed),
   990→    });
   991→    this.scene.launch('GameOver', { stats });
   992→    this.scene.pause();
   993→  }
