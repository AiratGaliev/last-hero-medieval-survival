    72→  create(payload?: { hero?: HeroId }): void {
    73→    this.resetState();
    74→    this.initHero(payload?.hero);
    75→    this.buildWorld();
    76→    this.buildPlayer();
    77→    this.buildSystems();
    78→    this.setupCollisions();

    80→    this.cameras.main.setBackgroundColor('#3a7bd5');
    81→    this.cameras.main.setBounds(0, 0, WORLD.SIZE, WORLD.SIZE);
    82→    this.cameras.main.startFollow(this.player, true, 0.12, 0.1);
    83→    this.resizeZoom();
    84→
    85→    this.scale.on('resize', this.resizeZoom, this);
    86→    this.events.once('shutdown', () => {
    87→      this.scale.off('resize', this.resizeZoom, this);
    88→      this.weapons?.destroy();
    89→      this.hud?.destroy();
    90→      this.pickups?.destroy();
    91→      audio.stopMusic();
    92→    });
    93→
    94→    audio.unlock();
    95→    audio.startMusic();
    96→    this.started = true;
    97→    this.hud.announce(`ВЫЖИВИ 15 МИНУТ, ${this.heroDef.name.toUpperCase()}!`, '#ffe9b0');

   114→  private initHero(heroId?: HeroId): void {
   115→    let id = heroId;
   116→    if (!id && typeof window !== 'undefined') {
   117→      const saved = window.localStorage.getItem('tss-hero');
   118→      if (saved === 'warrior' || saved === 'archer' || saved === 'lancer') id = saved;

   125→  private buildWorld(): void {
   126→    const S = WORLD.SIZE;
   127→    const m = WORLD.GRASS_MARGIN;
   128→    // Вода (фон)
   129→    const water = this.add.tileSprite(S / 2, S / 2, S, S, 'water-tile')
   130→      .setDepth(DEPTH.WATER - 20);
   131→    void water;
   132→    // Трава (арена)
   133→    const grass = this.add.tileSprite(S / 2, S / 2, S - m * 2, S - m * 2, 'grass-tile')
   134→      .setDepth(DEPTH.GROUND);
   135→    void grass;
   136→
   137→    // Анимированная пена по кромке воды
   138→    const foamStep = 192;
   139→    const foamAnim = this.anims.get('fx-foam');
   140→    const foamRing = (x: number, y: number) => {
   141→      const f = this.add.sprite(x, y, 'fx-foam').setDepth(DEPTH.WATER + 2);
   142→      f.play('fx-foam');
   143→      if (foamAnim) f.anims.setCurrentFrame(foamAnim.frames[Phaser.Math.Between(0, foamAnim.frames.length - 1)]);
   144→    };
   145→    for (let x = m; x < S - m; x += foamStep) {
   146→      foamRing(x + foamStep / 2, m);
   147→      foamRing(x + foamStep / 2, S - m);
   148→    }
   149→    for (let y = m; y < S - m; y += foamStep) {
   150→      foamRing(m, y + foamStep / 2);
   151→      foamRing(S - m, y + foamStep / 2);
   152→    }
   153→
   154→    this.physics.world.setBounds(m, m, S - m * 2, S - m * 2);
   155→
   156→    // Детерминированный декор
   157→    const rng = new Phaser.Math.RandomDataGenerator(['tiny-sword-survivors']);
   158→    const cx = S / 2, cy = S / 2;
   159→    const placed: { x: number; y: number; r: number }[] = [];
   160→    const canPlace = (x: number, y: number, r: number): boolean => {
   161→      if (Math.hypot(x - cx, y - cy) < 380) return false; // зона спавна
   162→      for (const p of placed) {
   163→        if (Math.hypot(x - p.x, y - p.y) < p.r + r) return false;
   164→      }
   165→      return true;
   166→    };
   167→    const place = (x: number, y: number, r: number) => placed.push({ x, y, r });
   168→    const randPos = (margin = 260): [number, number] => [
   169→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   170→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   171→    ];
   172→
   173→    // Замок — дом игрока
   174→    const castle = this.add.image(cx, cy - 460, 'castle').setScale(0.9)
   175→      .setDepth(DEPTH.ENTITY_BASE + cy - 460);
   176→    void castle;
   177→    place(cx, cy - 460, 200);
   178→    this.add.image(cx - 240, cy - 420, 'house2').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 420);
   179→    this.add.image(cx + 230, cy - 430, 'house1').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 430);
   180→    this.add.image(cx - 320, cy - 300, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 300);
   181→    this.add.image(cx + 320, cy - 280, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 280);
   182→    this.add.image(cx + 60, cy + 470, 'monastery').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy + 470);
   183→    place(cx + 60, cy + 470, 160);
   184→
   185→    // Дома и башни по карте
   186→    for (let i = 0; i < 8; i++) {
   187→      const [x, y] = randPos();
   188→      if (canPlace(x, y, 120)) {
   189→        const k = Phaser.Math.RND.pick(['house1', 'house2', 'house3']);
   190→        this.add.image(x, y, k).setScale(0.75).setDepth(DEPTH.ENTITY_BASE + y);
   191→        place(x, y, 120);
   192→      }
   193→    }
   194→    for (let i = 0; i < 6; i++) {
   195→      const [x, y] = randPos();
   196→      if (canPlace(x, y, 110)) {
   197→        this.add.image(x, y, 'tower').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + y);
   198→        place(x, y, 110);
   199→      }
   200→    }
   201→    // Деревья
   202→    for (let i = 0; i < 70; i++) {
   203→      const [x, y] = randPos();
   204→      if (!canPlace(x, y, 80)) continue;
   205→      const key = rng.frac() < 0.6 ? 'tree1' : 'tree3';
   206→      const sc = 0.8 + rng.frac() * 0.35;
   207→      const t = this.add.sprite(x, y, key).setScale(sc);
   208→      t.play(key === 'tree1' ? 'tree1-sway' : 'tree3-sway');
   209→      if (t.anims.currentAnim) t.setFrame(rng.between(0, 7));
   210→      t.setDepth(DEPTH.ENTITY_BASE + y);
   211→      place(x, y, 80);
   212→    }
   213→    // Камни и кусты
   214→    for (let i = 0; i < 30; i++) {
   215→      const [x, y] = randPos();
   216→      if (!canPlace(x, y, 40)) continue;
   217→      this.add.image(x, y, `rock${rng.between(1, 3)}`).setScale(0.8 + rng.frac() * 0.5)
   218→        .setDepth(DEPTH.ENTITY_BASE + y - 4);
   219→      place(x, y, 40);
   220→    }
   221→    for (let i = 0; i < 40; i++) {
   222→      const [x, y] = randPos();
   223→      if (!canPlace(x, y, 40)) continue;
   224→      const b = this.add.sprite(x, y, 'bushe1').setScale(0.7 + rng.frac() * 0.4);
   225→      b.play('bushe1-sway');
   226→      b.setFrame(rng.between(0, 7));
   227→      b.setDepth(DEPTH.ENTITY_BASE + y - 2);
   228→      place(x, y, 40);
   229→    }
   230→    // Овечки
   231→    for (let i = 0; i < 10; i++) {
   232→      const [x, y] = randPos();
   233→      if (!canPlace(x, y, 50)) continue;
   234→      const sh = this.add.sprite(x, y, 'sheep').setScale(0.8);
   235→      sh.play('sheep-idle');
   236→      sh.setDepth(DEPTH.ENTITY_BASE + y);
   237→      place(x, y, 50);
   238→    }
   239→    // Облака (дрейф вправо, относительный твин с переносом)
   240→    for (let i = 0; i < 10; i++) {
   241→      const [x, y] = [rng.between(0, S), rng.between(0, S)];
   242→      const c = this.add.image(x, y, `cloud${rng.between(1, 3)}`)
   243→        .setAlpha(0.5)
   244→        .setDepth(DEPTH.FX + 50);
   245→      this.tweens.add({
   246→        targets: c,
   247→        x: '+=700',
   248→        duration: rng.between(60, 100) * 1000,
   249→        repeat: -1,
   250→        onRepeat: () => { c.x = -320; },
   251→      });
   252→    }
   253→  }

   259→  private buildSystems(): void {
   260→    this.effects = new Effects(this);
   261→    this.pickups = new Pickups(
   262→      this,
   263→      (v) => this.addXp(v),
   264→      (v) => this.player.heal(v),
   265→      (x, y) => this.openChest(x, y),
   266→    );
   267→    this.projectiles = new Projectiles(this, {
   268→      onExplode: (x, y, dmg, radius) => this.explode(x, y, dmg, radius),
   269→    });
   270→    this.enemyGroup = this.physics.add.group();
   271→    this.enemyCtxStore = this.makeEnemyCtx();
   272→
   273→    const spawnCtx: SpawnerCtx = {
   274→      spawn: (kind, x, y) => this.spawnEnemy(kind, x, y),
   275→      spawnElite: (kind) => {
   276→        const a = Math.random() * Math.PI * 2;
   277→        this.spawnEnemy(kind, this.player.x + Math.cos(a) * 700, this.player.y + Math.sin(a) * 700);
   278→      },
   279→      spawnBoss: () => this.spawnBoss(),
   280→      playerX: () => this.player.x,
   281→      playerY: () => this.player.y,
   282→      announce: (t, c) => this.hud.announce(t, c),
   283→      spawnRadius: () => {
   284→        const cam = this.cameras.main;
   285→        return Math.hypot(cam.width, cam.height) / (2 * cam.zoom) + 80;
   286→      },
   287→    };
   288→    this.spawner = new Spawner(this, spawnCtx);
   289→
   290→    const weaponCtx: WeaponCtx = {
   291→      player: () => this.player,
   292→      enemies: () => this.activeEnemies(),
   293→      damageEnemy: (e, dmg, kx, ky, force) => this.damageEnemy(e, dmg, kx, ky, force),
   294→      projectiles: () => this.projectiles,
   295→      effects: () => this.effects,
   296→      moveDir: () => this.lastMove,
   297→      mightMult: () => this.mightMult,
   298→      cdMult: () => this.cdMult,
   299→    };
   300→    this.weapons = new WeaponSystem(this, weaponCtx);
   301→    this.weapons.add(this.heroDef.startWeapon);
   302→    const startId = this.heroDef.startWeapon;
   303→
   304→    this.hud = new HUD(this, () => this.openPause());
   305→    this.hud.setIcon(startId, WEAPONS[startId].icon, 1, false);
   306→  }

   308→  private setupCollisions(): void {
   309→    this.physics.add.overlap(this.player, this.enemyGroup, (_p, e) => {
   310→      const enemy = e as Enemy;
   311→      this.hurtPlayer(enemy.damage);
   312→    });
   313→
   314→    this.physics.add.overlap(this.projectiles.arrows, this.enemyGroup, (proj, e) => {
   315→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   316→      const enemy = e as Enemy;
   317→      if (!p.active || !enemy.active) return;
   318→      if (p.hitSet?.has(enemy)) return;
   319→      p.hitSet?.add(enemy);
   320→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 90);
   321→      this.projectiles.consumePierce(p);
   322→    });
   323→
   324→    this.physics.add.overlap(this.projectiles.spears, this.enemyGroup, (proj, e) => {
   325→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   326→      const enemy = e as Enemy;
   327→      if (!p.active || !enemy.active) return;
   328→      if (p.hitSet?.has(enemy)) return;
   329→      p.hitSet?.add(enemy);
   330→      this.damageEnemy(enemy, p.dmg, 0, 0, 40);
   331→      this.projectiles.consumePierce(p);
   332→    });
   333→
   334→    this.physics.add.overlap(this.projectiles.knives, this.enemyGroup, (proj, e) => {
   335→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   336→      const enemy = e as Enemy;
   337→      if (!p.active || !enemy.active) return;
   338→      if (p.hitSet?.has(enemy)) return;
   339→      p.hitSet?.add(enemy);
   340→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 70);
   341→      this.projectiles.consumePierce(p);
   342→    });
   343→
   344→    this.physics.add.overlap(this.player, this.projectiles.enemyArrows, (_p, proj) => {
   345→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   346→      if (!p.active) return;
   347→      this.projectiles.releaseEnemyArrow(p);
   348→      this.hurtPlayer(p.dmg);
   349→    });
   350→  }

   363→      esc: Phaser.Input.Keyboard.KeyCodes.ESC,

   373→  update(time: number, delta: number): void {
   374→    if (!this.started || this.gameEnded) return;
   375→    const dt = Math.min(delta, 50) / 1000;
   376→
   377→    // Пауза
   378→    if (Phaser.Input.Keyboard.JustDown(this.keys.esc) || Phaser.Input.Keyboard.JustDown(this.keys.p)) {
   379→      this.openPause();
   380→      return;
   381→    }
   382→    if (Phaser.Input.Keyboard.JustDown(this.keys.m)) {
   383→      const muted = audio.toggleMute();
   384→      this.hud.announce(muted ? 'ЗВУК ВЫКЛ' : 'ЗВУК ВКЛ', '#ffe9b0');
   385→    }
   386→
   387→    this.elapsed += dt;
   388→
   389→    // Движение игрока
   390→    let vx = 0, vy = 0;
   391→    if (this.keys.a.isDown || this.keys.left.isDown) vx -= 1;
   392→    if (this.keys.d.isDown || this.keys.right.isDown) vx += 1;
   393→    if (this.keys.w.isDown || this.keys.up.isDown) vy -= 1;
   394→    if (this.keys.s.isDown || this.keys.down.isDown) vy += 1;
   395→    const jv = this.joystick.vector;
   396→    vx += jv.x; vy += jv.y;
   397→    const len = Math.hypot(vx, vy);
   398→    if (len > 1) { vx /= len; vy /= len; }
   399→    this.player.setVelocity(vx * this.player.speed, vy * this.player.speed);
   400→    this.player.setAnimForVelocity();
   401→    if (len > 0.1) this.lastMove = { x: vx / Math.max(0.001, len), y: vy / Math.max(0.001, len) };
   402→    if (Math.abs(vx) > 0.05) this.player.setFlipX(vx < 0);
   403→
   404→    // Пыль под ногами
   405→    if (len > 0.5 && Math.random() < dt * 6) {
   406→      this.effects.dust(this.player.x, this.player.y + 14, 1);
   407→    }
   408→
   409→    // Системы
   410→    this.weapons.update(delta);
   411→    this.projectiles.update(delta);
   412→    this.spawner.update(dt);
   413→
   414→    // Сундуки с сокровищами
   415→    if (!this.gameEnded) {
   416→      this.chestTimer -= dt;
   417→      if (this.chestTimer <= 0) {
   418→        this.chestTimer = 75;
   419→        this.spawnChest();
   420→      }
   421→    }
   422→    this.pickups.update(delta, this.player.x, this.player.y, this.magnetRadius, time);
   423→
   424→    // Регенерация
   425→    if (this.regen > 0 && this.player.hp < this.player.maxHp) {
   426→      this.player.heal(this.regen * dt);
   427→    }
   428→
   429→    // Сепарация врагов
   430→    this.separateEnemies();
   431→
   432→    // Деспавн далёких
   433→    if (time - this.lastDespawn > 700) {
   434→      this.lastDespawn = time;
   435→      this.despawnFar();
   436→    }
   437→
   438→    // Босс
   439→    if (this.boss) {
   440→      if (this.boss.active) {
   441→        this.hud.updateBossBar(this.boss.hp / this.boss.maxHp);
   442→      } else {
   443→        this.boss = null;
   444→      }
   445→    }
   446→
   447→    // HUD
   448→    if (time - this.lastHudUpdate > 150) {
   449→      this.lastHudUpdate = time;
   450→      this.hud.update({
   451→        hp: this.player.hp, maxHp: this.player.maxHp,
   452→        level: this.level, xp: this.xp, xpNext: this.xpNext,
   453→        kills: this.kills, score: this.currentScore(), elapsed: this.elapsed, gold: this.gold,
   454→      });
   455→    }
   456→
   457→    // Левел-ап
   458→    if (this.pendingLevelUps > 0 && !this.overlayOpen && !this.gameEnded) {
   459→      this.openLevelUp();
   460→    }
   461→  }

   469→  private enemiesNear(x: number, y: number, r: number): Phaser.Physics.Arcade.Sprite[] {
   470→    const out: Phaser.Physics.Arcade.Sprite[] = [];
   471→    for (const e of this.enemyGroup.getChildren() as Enemy[]) {

   477→  private damageEnemy(e: Enemy, dmg: number, kx = 0, ky = 0, force = 0): void {
   478→    if (!e.active) return;
   479→    let final = dmg * this.mightMult;
   480→    const crit = Math.random() < this.critChance;
   481→    if (crit) final *= 2;
   482→    final = Math.max(1, Math.round(final));
   483→    this.effects.damage(e.x, e.y - 40 * e.def.scale, final, crit);
   484→    if (force > 0) e.knockback(kx, ky, force);
   485→    this.effects.hitSound(this.time.now);
   486→    if (e.takeDamage(final)) {
   487→      this.onEnemyDeath(e);
   488→    }
   489→  }

   598→  private separateEnemies(): void {
   599→    this.sepGrid.clear();
   600→    const cell = 64;
   601→    const px = this.player.x, py = this.player.y;
   602→    const near: Enemy[] = [];
   603→    for (const e of this.enemyGroup.getChildren() as Enemy[]) {
   604→      if (!e.active) continue;
   605→      if (Math.abs(e.x - px) > 900 || Math.abs(e.y - py) > 700) continue;
   606→      near.push(e);
   607→      const key = (Math.floor(e.x / cell) << 16) ^ Math.floor(e.y / cell);
   608→      const arr = this.sepGrid.get(key);
   609→      if (arr) arr.push(e);
   610→      else this.sepGrid.set(key, [e]);
   611→    }
   612→    for (const e of near) {
   613→      const cx = Math.floor(e.x / cell), cy = Math.floor(e.y / cell);
   614→      for (let gx = cx - 1; gx <= cx + 1; gx++) {
   615→        for (let gy = cy - 1; gy <= cy + 1; gy++) {
   616→          const arr = this.sepGrid.get((gx << 16) ^ gy);
   617→          if (!arr) continue;
   618→          for (const o of arr) {
   619→            if (o === e || o.x < e.x) continue; // каждая пара один раз
   620→            const dx = o.x - e.x, dy = o.y - e.y;
   621→            const d = Math.hypot(dx, dy);
   622→            const min = (e.def.bodyRadius + o.def.bodyRadius) * 0.85;
   623→            if (d > 0.001 && d < min) {
   624→              const push = (min - d) / 2;
   625→              const nx = dx / d, ny = dy / d;
   626→              e.x -= nx * push; e.y -= ny * push;
   627→              o.x += nx * push; o.y += ny * push;
   628→            }
   629→          }
   630→        }
   631→      }
   632→    }
   633→  }

   656→  private generateChoices(): UpgradeCard[] {
   657→    const pool: UpgradeCard[] = [];
   658→    for (const w of Object.values(WEAPONS)) {
   659→      const lvl = this.weapons.level(w.id);
   660→      if (lvl === 0 && this.weapons.ownedIds().length < 7) {
   661→        pool.push({
   662→          kind: 'weapon-new', id: w.id, title: w.name, subtitle: 'НОВОЕ ОРУЖИЕ',
   663→          desc: w.desc, icon: w.icon, levelFrom: 0, levelTo: 1,
   664→        });
   665→      } else if (lvl > 0 && lvl < 5) {
   666→        const next = w.levels[lvl];
   667→        pool.push({
   668→          kind: 'weapon-up', id: w.id, title: w.name, subtitle: `УРОВЕНЬ ${lvl} → ${lvl + 1}`,
   669→          desc: this.weaponUpgradeDesc(w.id, lvl), icon: w.icon, levelFrom: lvl, levelTo: lvl + 1,
   670→        });
   671→        void next;
   672→      }
   673→    }
   674→    for (const p of Object.values(PASSIVES)) {
   675→      const lvl = this.passiveLevels.get(p.id) ?? 0;
   676→      if (lvl < p.max) {
   677→        pool.push({
   678→          kind: 'passive', id: p.id, title: p.name, subtitle: lvl === 0 ? 'НОВЫЙ БОНУС' : `УРОВЕНЬ ${lvl} → ${lvl + 1}`,
   679→          desc: p.desc(lvl + 1), icon: p.icon, levelFrom: lvl, levelTo: lvl + 1,
   680→        });
   681→      }
   682→    }
   683→    Phaser.Utils.Array.Shuffle(pool);
   684→    const picked = pool.slice(0, 3);
   685→    if (picked.length === 0) {
   686→      picked.push(
   687→        { kind: 'heal', title: 'Жареное мясо', subtitle: 'ЛЕЧЕНИЕ', desc: 'Восстановить 40 HP', icon: 'icon-meat' },
   688→        { kind: 'gold', title: 'Кошель золота', subtitle: 'ЗОЛОТО', desc: '+100 золота к счёту', icon: 'icon-coin' },
   689→        { kind: 'heal', title: 'Жареное мясо', subtitle: 'ЛЕЧЕНИЕ', desc: 'Восстановить 40 HP', icon: 'icon-meat' },
   690→      );
   691→    }
   692→    while (picked.length < 3) {
   693→      picked.push({ kind: 'heal', title: 'Жареное мясо', subtitle: 'ЛЕЧЕНИЕ', desc: 'Восстановить 40 HP', icon: 'icon-meat' });
   694→    }
   695→    return picked;
   696→  }

   698→  private weaponUpgradeDesc(id: WeaponId, lvl: number): string {
   699→    const w = WEAPONS[id];
   700→    const cur = w.levels[lvl - 1];
   701→    const next = w.levels[lvl];
   702→    const parts: string[] = [`Урон ${cur.dmg} → ${next.dmg}`];
   703→    if (next.count && next.count !== cur.count) parts.push(`Количество: ${next.count}`);
   704→    if (next.radius && cur.radius && next.radius !== cur.radius) parts.push(`Радиус: ${next.radius}`);
   705→    if (next.pierce && cur.pierce && next.pierce !== cur.pierce) parts.push(`Пробитие: ${next.pierce}`);
   706→    return parts.join(', ');
   707→  }

   728→  private applyCard(card: UpgradeCard): void {
   729→    switch (card.kind) {
   730→      case 'weapon-new': {
   731→        const id = card.id as WeaponId;
   732→        this.weapons.add(id);
   733→        this.hud.setIcon(id, WEAPONS[id].icon, this.weapons.level(id), false);
   734→        break;
   735→      }
   736→      case 'weapon-up': {
   737→        const id = card.id as WeaponId;
   738→        this.weapons.upgrade(id);
   739→        this.hud.setIcon(id, WEAPONS[id].icon, this.weapons.level(id), false);
   740→        break;
   741→      }
   742→      case 'passive': {
   743→        const id = card.id as PassiveId;
   744→        const lvl = (this.passiveLevels.get(id) ?? 0) + 1;
   745→        this.passiveLevels.set(id, lvl);
   746→        this.hud.setIcon(id, PASSIVES[id].icon, lvl, true);
   747→        this.recomputeStats();
   748→        if (id === 'vitality') this.player.heal(25);
   749→        break;
   750→      }
   751→      case 'heal': this.player.heal(40); break;
   752→      case 'gold': this.gold += 100; break;
   753→    }
   754→  }

   756→  private recomputeStats(): void {
   757→    const lvl = (id: PassiveId) => this.passiveLevels.get(id) ?? 0;
   758→    this.player.speed = this.heroDef.speed * (1 + 0.1 * lvl('boots'));
   759→    this.armorMult = 1 - 0.08 * lvl('armor');
   760→    this.magnetRadius = PLAYER_BASE.MAGNET_RADIUS * (1 + 0.3 * lvl('magnet'));
   761→    this.regen = 0.5 * lvl('regen');
   762→    this.mightMult = 1 + 0.12 * lvl('might');
   763→    this.cdMult = Math.max(0.6, 1 - 0.08 * lvl('haste'));
   764→    this.critChance = 0.15 * lvl('luck');
   765→    this.xpMult = 1 + 0.15 * lvl('wisdom');
   766→    const newMax = this.heroDef.maxHp + 25 * lvl('vitality');
   767→    if (newMax !== this.player.maxHp) {
   768→      const diff = newMax - this.player.maxHp;
   769→      this.player.maxHp = newMax;
   770→      this.player.hp = Math.min(newMax, this.player.hp + Math.max(0, diff));
   771→    }
   772→  }
