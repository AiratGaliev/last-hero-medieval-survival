   154→  private buildWorld(): void {
   155→    const S = WORLD.SIZE;
   156→    const m = WORLD.GRASS_MARGIN;
   157→    // Вода (фон)
   158→    const water = this.add.tileSprite(S / 2, S / 2, S, S, 'water-tile')
   159→      .setDepth(DEPTH.WATER - 20);
   160→    void water;
   161→    // Трава (арена)
   162→    const grass = this.add.tileSprite(S / 2, S / 2, S - m * 2, S - m * 2, 'grass-tile')
   163→      .setDepth(DEPTH.GROUND);
   164→    void grass;
   165→
   166→    // Анимированная пена по кромке воды
   167→    const foamStep = 192;
   168→    const foamAnim = this.anims.get('fx-foam');
   169→    const foamRing = (x: number, y: number) => {
   170→      const f = this.add.sprite(x, y, 'fx-foam').setDepth(DEPTH.WATER + 2);
   171→      f.play('fx-foam');
   172→      if (foamAnim) f.anims.setCurrentFrame(foamAnim.frames[Phaser.Math.Between(0, foamAnim.frames.length - 1)]);
   173→    };
   174→    for (let x = m; x < S - m; x += foamStep) {
   175→      foamRing(x + foamStep / 2, m);
   176→      foamRing(x + foamStep / 2, S - m);
   177→    }
   178→    for (let y = m; y < S - m; y += foamStep) {
   179→      foamRing(m, y + foamStep / 2);
   180→      foamRing(S - m, y + foamStep / 2);
   181→    }
   182→
   183→    this.physics.world.setBounds(m, m, S - m * 2, S - m * 2);
   184→
   185→    // Детерминированный декор
   186→    const rng = new Phaser.Math.RandomDataGenerator(['tiny-sword-survivors']);
   187→    const cx = S / 2, cy = S / 2;
   188→    const placed: { x: number; y: number; r: number }[] = [];
   189→    const canPlace = (x: number, y: number, r: number): boolean => {
   190→      if (Math.hypot(x - cx, y - cy) < 380) return false; // зона спавна
   191→      for (const p of placed) {
   192→        if (Math.hypot(x - p.x, y - p.y) < p.r + r) return false;
   193→      }
   194→      return true;
   195→    };
   196→    const place = (x: number, y: number, r: number) => placed.push({ x, y, r });
   197→    const randPos = (margin = 260): [number, number] => [
   198→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   199→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   200→    ];
   201→
   202→    // Замок — дом игрока
   203→    const castle = this.add.image(cx, cy - 460, 'castle').setScale(0.9)
   204→      .setDepth(DEPTH.ENTITY_BASE + cy - 460);
   205→    void castle;
   206→    place(cx, cy - 460, 200);
   207→    this.add.image(cx - 240, cy - 420, 'house2').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 420);
   208→    this.add.image(cx + 230, cy - 430, 'house1').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 430);
   209→    this.add.image(cx - 320, cy - 300, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 300);
   210→    this.add.image(cx + 320, cy - 280, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 280);
   211→    this.add.image(cx + 60, cy + 470, 'monastery').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy + 470);
   212→    place(cx + 60, cy + 470, 160);
   213→
   214→    // Дома и башни по карте
   215→    for (let i = 0; i < 8; i++) {
   216→      const [x, y] = randPos();
   217→      if (canPlace(x, y, 120)) {
   218→        const k = Phaser.Math.RND.pick(['house1', 'house2', 'house3']);
   219→        this.add.image(x, y, k).setScale(0.75).setDepth(DEPTH.ENTITY_BASE + y);
   220→        place(x, y, 120);
   221→      }
   222→    }
   223→    for (let i = 0; i < 6; i++) {
   224→      const [x, y] = randPos();
   225→      if (canPlace(x, y, 110)) {
   226→        this.add.image(x, y, 'tower').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + y);
   227→        place(x, y, 110);
   228→      }
   229→    }
   230→    // Деревья
   231→    for (let i = 0; i < 70; i++) {
   232→      const [x, y] = randPos();
   233→      if (!canPlace(x, y, 80)) continue;
   234→      const key = rng.frac() < 0.6 ? 'tree1' : 'tree3';
   235→      const sc = 0.8 + rng.frac() * 0.35;
   236→      const t = this.add.sprite(x, y, key).setScale(sc);
   237→      t.play(key === 'tree1' ? 'tree1-sway' : 'tree3-sway');
   238→      if (t.anims.currentAnim) t.setFrame(rng.between(0, 7));
   239→      t.setDepth(DEPTH.ENTITY_BASE + y);
   240→      place(x, y, 80);
   241→    }
   242→    // Камни и кусты
   243→    for (let i = 0; i < 30; i++) {
   244→      const [x, y] = randPos();
   245→      if (!canPlace(x, y, 40)) continue;
   246→      this.add.image(x, y, `rock${rng.between(1, 3)}`).setScale(0.8 + rng.frac() * 0.5)
   247→        .setDepth(DEPTH.ENTITY_BASE + y - 4);
   248→      place(x, y, 40);
   249→    }
   250→    for (let i = 0; i < 40; i++) {
   251→      const [x, y] = randPos();
   252→      if (!canPlace(x, y, 40)) continue;
   253→      const b = this.add.sprite(x, y, 'bushe1').setScale(0.7 + rng.frac() * 0.4);
   254→      b.play('bushe1-sway');
   255→      b.setFrame(rng.between(0, 7));
   256→      b.setDepth(DEPTH.ENTITY_BASE + y - 2);
   257→      place(x, y, 40);
   258→    }
   259→    // Овечки
   260→    for (let i = 0; i < 10; i++) {
   261→      const [x, y] = randPos();
   262→      if (!canPlace(x, y, 50)) continue;
   263→      const sh = this.add.sprite(x, y, 'sheep').setScale(0.8);
   264→      sh.play('sheep-idle');
   265→      sh.setDepth(DEPTH.ENTITY_BASE + y);
   266→      place(x, y, 50);
   267→    }
   268→    // Облака (дрейф вправо, относительный твин с переносом)
   269→    for (let i = 0; i < 10; i++) {
   270→      const [x, y] = [rng.between(0, S), rng.between(0, S)];
   271→      const c = this.add.image(x, y, `cloud${rng.between(1, 3)}`)
   272→        .setAlpha(0.5)
   273→        .setDepth(DEPTH.FX + 50);
   274→      this.tweens.add({
   275→        targets: c,
   276→        x: '+=700',
   277→        duration: rng.between(60, 100) * 1000,
   278→        repeat: -1,
   279→        onRepeat: () => { c.x = -320; },
   280→      });
   281→    }
   282→  }

   342→  private setupCollisions(): void {
   343→    this.physics.add.overlap(this.player, this.enemyGroup, (_p, e) => {
   344→      const enemy = e as Enemy;
   345→      this.hurtPlayer(enemy.damage);
   346→    });
   347→
   348→    this.physics.add.overlap(this.projectiles.arrows, this.enemyGroup, (proj, e) => {
   349→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   350→      const enemy = e as Enemy;
   351→      if (!p.active || !enemy.active) return;
   352→      if (p.hitSet?.has(enemy)) return;
   353→      p.hitSet?.add(enemy);
   354→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 90);
   355→      this.projectiles.consumePierce(p);
   356→    });
   357→
   358→    this.physics.add.overlap(this.projectiles.spears, this.enemyGroup, (proj, e) => {
   359→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   360→      const enemy = e as Enemy;
   361→      if (!p.active || !enemy.active) return;
   362→      if (p.hitSet?.has(enemy)) return;
   363→      p.hitSet?.add(enemy);
   364→      this.damageEnemy(enemy, p.dmg, 0, 0, 40);
   365→      this.projectiles.consumePierce(p);
   366→    });
   367→
   368→    this.physics.add.overlap(this.projectiles.knives, this.enemyGroup, (proj, e) => {
   369→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   370→      const enemy = e as Enemy;
   371→      if (!p.active || !enemy.active) return;
   372→      if (p.hitSet?.has(enemy)) return;
   373→      p.hitSet?.add(enemy);
   374→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 70);
   375→      this.projectiles.consumePierce(p);
   376→    });
   377→
   378→    this.physics.add.overlap(this.player, this.projectiles.enemyArrows, (_p, proj) => {
   379→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   380→      if (!p.active) return;
   381→      this.projectiles.releaseEnemyArrow(p);
   382→      this.hurtPlayer(p.dmg);
   383→    });

   407→  update(time: number, delta: number): void {
   408→    if (!this.started || this.gameEnded) return;
   409→
   410→    // Хитстоп: короткая заморозка мира (после убийств элит/взрывов/урона)
   411→    if (time < this.hitstopUntil) {
   412→      this.player.setVelocity(0, 0);
   413→      return;
   414→    }
   415→
   416→    const dt = Math.min(delta, 50) / 1000;
   417→
   418→    // Пауза
   419→    if (Phaser.Input.Keyboard.JustDown(this.keys.esc) || Phaser.Input.Keyboard.JustDown(this.keys.p)) {
   420→      this.openPause();
   421→      return;
   422→    }
   423→    if (Phaser.Input.Keyboard.JustDown(this.keys.m)) {
   424→      const muted = audio.toggleMute();
   425→      this.hud.announce(muted ? 'ЗВУК ВЫКЛ' : 'ЗВУК ВКЛ', '#ffe9b0');
   426→    }
   427→
   428→    this.elapsed += dt;
   429→
   430→    // Движение игрока
   431→    let vx = 0, vy = 0;
   432→    if (this.keys.a.isDown || this.keys.left.isDown) vx -= 1;
   433→    if (this.keys.d.isDown || this.keys.right.isDown) vx += 1;
   434→    if (this.keys.w.isDown || this.keys.up.isDown) vy -= 1;
   435→    if (this.keys.s.isDown || this.keys.down.isDown) vy += 1;
   436→    const jv = this.joystick.vector;
   437→    vx += jv.x; vy += jv.y;
   438→    const len = Math.hypot(vx, vy);
   439→    if (len > 1) { vx /= len; vy /= len; }
   440→    this.player.setVelocity(vx * this.player.speed, vy * this.player.speed);
   441→    this.player.setAnimForVelocity();
   442→    if (len > 0.1) this.lastMove = { x: vx / Math.max(0.001, len), y: vy / Math.max(0.001, len) };
   443→    if (Math.abs(vx) > 0.05) this.player.setFlipX(vx < 0);
   444→
   445→    // Пыль под ногами
   446→    if (len > 0.5 && Math.random() < dt * 6) {
   447→      this.effects.dust(this.player.x, this.player.y + 14, 1);
   448→    }
   449→
   450→    // Системы
   451→    this.weapons.update(delta);
   452→    this.projectiles.update(delta);
   453→    this.spawner.update(dt);
   454→
   455→    // Сундуки с сокровищами
   456→    if (!this.gameEnded) {
   457→      this.chestTimer -= dt;
   458→      if (this.chestTimer <= 0) {
   459→        this.chestTimer = 75;
   460→        this.spawnChest();
   461→      }
   462→    }
   463→    this.pickups.update(delta, this.player.x, this.player.y, this.magnetRadius, time);
   464→
   465→    // Регенерация
   466→    if (this.regen > 0 && this.player.hp < this.player.maxHp) {
   467→      this.player.heal(this.regen * dt);
   468→    }
   469→
   470→    // Сепарация врагов
   471→    this.separateEnemies();
   472→
   473→    // Деспавн далёких
   474→    if (time - this.lastDespawn > 700) {
   475→      this.lastDespawn = time;
   476→      this.despawnFar();
   477→    }
   478→
   479→    // Босс / мидбосс — приоритет полосе HP: сначала финальный босс
   480→    if (this.boss) {
   481→      if (this.boss.active) {
   482→        this.hud.updateBossBar(this.boss.hp / this.boss.maxHp);
   483→      } else {
   484→        this.boss = null;
   485→        if (this.midboss && this.midboss.active) this.hud.showBossBar(ENEMIES.midboss.name);
   486→        else this.hud.hideBossBar();
   487→      }
   488→    } else if (this.midboss) {
   489→      if (this.midboss.active) {
   490→        this.hud.updateBossBar(this.midboss.hp / this.midboss.maxHp);
   491→      } else {
   492→        this.midboss = null;
   493→        this.hud.hideBossBar();
   494→      }
   495→    }
   496→
   497→    // HUD
   498→    if (time - this.lastHudUpdate > 150) {
   499→      this.lastHudUpdate = time;
   500→      this.hud.update({
   501→        hp: this.player.hp, maxHp: this.player.maxHp,
   502→        level: this.level, xp: this.xp, xpNext: this.xpNext,
   503→        kills: this.kills, score: this.currentScore(), elapsed: this.elapsed, gold: this.gold,
   504→      });
   505→    }
   506→
   507→    // Левел-ап
   508→    if (this.pendingLevelUps > 0 && !this.overlayOpen && !this.gameEnded) {
   509→      this.openLevelUp();
   510→    }
   511→  }

   533→  private damageEnemy(e: Enemy, dmg: number, kx = 0, ky = 0, force = 0): void {
   534→    if (!e.active || e.dead) return;
   535→    let final = dmg * this.mightMult;
   536→    const crit = Math.random() < this.critChance;
   537→    if (crit) final *= 2;
   538→    final = Math.max(1, Math.round(final));
   539→    this.effects.damage(e.x, e.y - 40 * e.def.scale, final, crit);
   540→    if (Math.random() < 0.6) this.effects.hitSpark(e.x, e.y - 26 * e.def.scale);
   541→    if (force > 0) e.knockback(kx, ky, force);
   542→    this.effects.hitSound(this.time.now);
   543→    if (e.takeDamage(final)) {
   544→      this.onEnemyDeath(e);
   545→    }
   546→  }

   573→  private onEnemyDeath(e: Enemy): void {
   574→    if (!e.active) return;
   575→    // Убираем из мира СРАЗУ: цепные взрывы не должны повторно бить погибающего
   576→    const x = e.x, y = e.y;
   577→    this.enemyGroup.remove(e, true, true);
   578→
   579→    this.kills++;
   580→    this.score += SCORE.KILL;
   581→    this.ach?.event({ type: 'kills', value: this.kills });
   582→    const big = e.def.isElite || e.def.isBoss || e.def.isMidboss;
   583→
   584→    // Бочка-камикадзе, убитая оружием: взрыв бьёт врагов (цепная реакция!)
   585→    if (e.def.kind === 'bomber') {
   586→      this.bomberExplosion(x, y, false);
   587→    }
   588→
   589→    this.effects.explosion(x, y, big ? 1.7 : e.def.scale * 1.15, big ? 2 : 1);
   590→    if (big) this.hitstop(110);
   591→    if (this.time.now - this.lastDieSound > 120) {
   592→      this.lastDieSound = this.time.now;
   593→      audio.play('die');
   594→    }
   595→    // Дроп
   596→    if (e.def.gem && e.xpValue > 0) {
   597→      const n = big ? 3 : 1;
   598→      for (let i = 0; i < n; i++) {
   599→        const ox = big ? Phaser.Math.Between(-40, 40) : 0;
   600→        const oy = big ? Phaser.Math.Between(-40, 40) : 0;
   601→        this.pickups.spawnGem(x + ox, y + oy, e.def.gem, Math.ceil(e.xpValue / n));
   602→      }
   603→    }
   604→    if (e.def.kind === 'brute' && Math.random() < 0.04) this.pickups.spawnMeat(x, y);
   605→    if (e.def.isElite) {
   606→      this.pickups.spawnMeat(x + 30, y);
   607→      this.pickups.spawnMeat(x - 30, y);
   608→      this.hud.announce('ЭЛИТА ПОВЕРЖЕНА! +25 XP', '#7CFC00');
   609→      this.elitesSlain++;
   610→      this.ach?.event({ type: 'elites', value: this.elitesSlain });
   611→    }
   612→    // Золотые кристаллы: теперь дают золото
   613→    if (Math.random() < 0.03) this.pickups.spawnGoldGem(x, y, Phaser.Math.Between(3, 8));
   614→    if (e.def.kind === 'midboss') {
   615→      this.score += 1500;
   616→      this.pickups.spawnGoldGem(x + 24, y, 40);
   617→      this.pickups.spawnGoldGem(x - 24, y, 40);
   618→      for (let i = 0; i < 3; i++) {
   619→        this.pickups.spawnGem(x + Phaser.Math.Between(-50, 50), y + Phaser.Math.Between(-50, 50), 'gem-red', 12);
   620→      }
   621→      this.hud.announce('ТЁМНЫЙ МОНАХ ПОВЕРЖЕН! +1500 ОЧКОВ', '#ffd700');
   622→    }
   623→
   624→    if (e === this.boss) {
   625→      this.victory();
   626→    }
   627→  }

   718→  private separateEnemies(): void {
   719→    this.sepGrid.clear();
   720→    const cell = 64;
   721→    const px = this.player.x, py = this.player.y;
   722→    const near: Enemy[] = [];
   723→    for (const e of this.enemyGroup.getChildren() as Enemy[]) {
   724→      if (!e.active) continue;
   725→      if (Math.abs(e.x - px) > 900 || Math.abs(e.y - py) > 700) continue;
   726→      near.push(e);
   727→      const key = (Math.floor(e.x / cell) << 16) ^ Math.floor(e.y / cell);
   728→      const arr = this.sepGrid.get(key);
   729→      if (arr) arr.push(e);
   730→      else this.sepGrid.set(key, [e]);
   731→    }
   732→    for (const e of near) {
   733→      const cx = Math.floor(e.x / cell), cy = Math.floor(e.y / cell);
   734→      for (let gx = cx - 1; gx <= cx + 1; gx++) {
   735→        for (let gy = cy - 1; gy <= cy + 1; gy++) {
   736→          const arr = this.sepGrid.get((gx << 16) ^ gy);
   737→          if (!arr) continue;
   738→          for (const o of arr) {
   739→            if (o === e || o.x < e.x) continue; // каждая пара один раз
   740→            const dx = o.x - e.x, dy = o.y - e.y;
   741→            const d = Math.hypot(dx, dy);
   742→            const min = (e.def.bodyRadius + o.def.bodyRadius) * 0.85;
   743→            if (d > 0.001 && d < min) {
   744→              const push = (min - d) / 2;
   745→              const nx = dx / d, ny = dy / d;
   746→              e.x -= nx * push; e.y -= ny * push;
   747→              o.x += nx * push; o.y += ny * push;
   748→            }
   749→          }
   750→        }
   751→      }
   752→    }
   753→  }

   828→  private weaponUpgradeDesc(id: WeaponId, lvl: number): string {
   829→    const w = WEAPONS[id];
   830→    const cur = w.levels[lvl - 1];
   831→    const next = w.levels[lvl];
   832→    const parts: string[] = [`Урон ${cur.dmg} → ${next.dmg}`];
   833→    if (next.count && next.count !== cur.count) parts.push(`Количество: ${next.count}`);
   834→    if (next.radius && cur.radius && next.radius !== cur.radius) parts.push(`Радиус: ${next.radius}`);
   835→    if (next.pierce && cur.pierce && next.pierce !== cur.pierce) parts.push(`Пробитие: ${next.pierce}`);
   836→    return parts.join(', ');

   859→    switch (card.kind) {

   863→        this.hud.setIcon(id, WEAPONS[id].icon, this.weapons.level(id), false);

   869→        this.hud.setIcon(id, WEAPONS[id].icon, this.weapons.level(id), false);
   870→        break;
   871→      }
   872→      case 'passive': {
   873→        const id = card.id as PassiveId;

   877→        this.recomputeStats();
   878→        if (id === 'vitality') this.player.heal(25);
   879→        break;
   880→      }
   881→      case 'heal': this.player.heal(40); break;
   882→      case 'gold':
   883→        this.gold += 100;
   884→        this.ach?.event({ type: 'gold', value: this.gold });
   885→        break;

   889→  private recomputeStats(): void {
   890→    const lvl = (id: PassiveId) => this.passiveLevels.get(id) ?? 0;
   891→    const shop = this.shopLevels;
   892→    this.player.speed = this.heroDef.speed * (1 + 0.04 * shop.speed + 0.1 * lvl('boots'));
   893→    this.armorMult = 1 - 0.08 * lvl('armor');
   894→    this.magnetRadius = PLAYER_BASE.MAGNET_RADIUS * (1 + 0.2 * shop.magnet + 0.3 * lvl('magnet'));
   895→    this.regen = this.heroDef.baseRegen + 0.5 * lvl('regen');
   896→    this.mightMult = 1 + 0.06 * shop.might + 0.12 * lvl('might');
   897→    this.cdMult = Math.max(0.6, 1 - 0.08 * lvl('haste'));
   898→    this.critChance = 0.05 * shop.luck + 0.15 * lvl('luck');
   899→    this.xpMult = 1 + 0.15 * lvl('wisdom');
   900→    const newMax = this.heroDef.maxHp + 15 * shop.vitality + 25 * lvl('vitality');
   901→    if (newMax !== this.player.maxHp) {
   902→      const diff = newMax - this.player.maxHp;
   903→      this.player.maxHp = newMax;
   904→      this.player.hp = Math.min(newMax, this.player.hp + Math.max(0, diff));
   905→    }
   906→  }

   924→  private currentScore(): number {
   925→    const base = this.score
   926→      + Math.floor(this.elapsed / 60) * SCORE.MINUTE
   927→      + this.level * SCORE.LEVEL
   928→      + this.gold;
   929→    return Math.round(base * this.diffDef.scoreMult);
   930→  }

   932→  private victory(): void {
   933→    if (this.gameEnded) return;
   934→    this.score += SCORE.BOSS;
   935→    this.hud.hideBossBar();
   936→    this.hud.announce('ПОБЕДА!!! БОСС ПОВЕРЖЕН!', '#ffd700');
   937→    audio.play('victory');
   938→    this.cameras.main.shake(500, 0.006);
   939→    this.hitstop(200);
   940→    this.ach?.event({ type: 'victory', difficulty: this.diffDef.id });
   941→    // Финальный фейерверк
   942→    for (let i = 0; i < 8; i++) {
   943→      this.time.delayedCall(i * 160, () => {
   944→        this.effects.explosion(
   945→          this.player.x + Phaser.Math.Between(-300, 300),
   946→          this.player.y + Phaser.Math.Between(-300, 300), 1.6, 2,
   947→        );
   948→      });
   949→    }
   950→    // Собрать все кристаллы
   951→    this.pickups.vacuumAll(this.player.x, this.player.y);
   952→    this.time.delayedCall(1900, () => this.gameOver(true));
   953→  }

   955→  private gameOver(victory: boolean): void {
   956→    if (this.gameEnded) return;
   957→    this.gameEnded = true;
   958→    this.spawner.stopped = true;
   959→    this.player.setVelocity(0, 0);
   960→    if (!victory) audio.play('gameover');
   961→    audio.stopMusic();
   962→
   963→    const finalScore = this.currentScore();
   964→    const stats = {
   965→      victory,
   966→      time: Math.floor(this.elapsed),
   967→      kills: this.kills,
   968→      level: this.level,
   969→      gold: this.gold,
   970→      score: finalScore,
   971→      hero: this.heroDef.name,
   972→      difficulty: this.diffDef.name,
   973→      newAch: (this.ach?.newlyUnlocked ?? []).map((a) => ({ name: a.name, emoji: a.emoji })),
   974→    };
   975→    // Мета-статистика (localStorage) для React-сайдбара
   976→    commitMeta({
   977→      victory,
   978→      kills: this.kills,
   979→      gold: this.gold,
   980→      score: finalScore,
   981→      playTime: Math.floor(this.elapsed),
   982→    });
   983→    this.scene.launch('GameOver', { stats });
   984→    this.scene.pause();
   985→  }
