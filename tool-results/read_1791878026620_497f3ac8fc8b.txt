   125→   125→  private buildWorld(): void {
   126→   126→    const S = WORLD.SIZE;
   127→   127→    const m = WORLD.GRASS_MARGIN;
   128→   128→    // Вода (фон)
   129→   129→    const water = this.add.tileSprite(S / 2, S / 2, S, S, 'water-tile')
   130→   130→      .setDepth(DEPTH.WATER - 20);
   131→   131→    void water;
   132→   132→    // Трава (арена)
   133→   133→    const grass = this.add.tileSprite(S / 2, S / 2, S - m * 2, S - m * 2, 'grass-tile')
   134→   134→      .setDepth(DEPTH.GROUND);
   135→   135→    void grass;
   136→   136→
   137→   137→    // Анимированная пена по кромке воды
   138→   138→    const foamStep = 192;
   139→   139→    const foamAnim = this.anims.get('fx-foam');
   140→   140→    const foamRing = (x: number, y: number) => {
   141→   141→      const f = this.add.sprite(x, y, 'fx-foam').setDepth(DEPTH.WATER + 2);
   142→   142→      f.play('fx-foam');
   143→   143→      if (foamAnim) f.anims.setCurrentFrame(foamAnim.frames[Phaser.Math.Between(0, foamAnim.frames.length - 1)]);
   144→   144→    };
   145→   145→    for (let x = m; x < S - m; x += foamStep) {
   146→   146→      foamRing(x + foamStep / 2, m);
   147→   147→      foamRing(x + foamStep / 2, S - m);
   148→   148→    }
   149→   149→    for (let y = m; y < S - m; y += foamStep) {
   150→   150→      foamRing(m, y + foamStep / 2);
   151→   151→      foamRing(S - m, y + foamStep / 2);
   152→   152→    }
   153→   153→
   154→   154→    this.physics.world.setBounds(m, m, S - m * 2, S - m * 2);
   155→   155→
   156→   156→    // Детерминированный декор
   157→   157→    const rng = new Phaser.Math.RandomDataGenerator(['tiny-sword-survivors']);
   158→   158→    const cx = S / 2, cy = S / 2;
   159→   159→    const placed: { x: number; y: number; r: number }[] = [];
   160→   160→    const canPlace = (x: number, y: number, r: number): boolean => {
   161→   161→      if (Math.hypot(x - cx, y - cy) < 380) return false; // зона спавна
   162→   162→      for (const p of placed) {
   163→   163→        if (Math.hypot(x - p.x, y - p.y) < p.r + r) return false;
   164→   164→      }
   165→   165→      return true;
   166→   166→    };
   167→   167→    const place = (x: number, y: number, r: number) => placed.push({ x, y, r });
   168→   168→    const randPos = (margin = 260): [number, number] => [
   169→   169→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   170→   170→      Phaser.Math.Clamp(rng.between(m + 80, S - m - 80), margin, S - margin),
   171→   171→    ];
   172→   172→
   173→   173→    // Замок — дом игрока
   174→   174→    const castle = this.add.image(cx, cy - 460, 'castle').setScale(0.9)
   175→   175→      .setDepth(DEPTH.ENTITY_BASE + cy - 460);
   176→   176→    void castle;
   177→   177→    place(cx, cy - 460, 200);
   178→   178→    this.add.image(cx - 240, cy - 420, 'house2').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 420);
   179→   179→    this.add.image(cx + 230, cy - 430, 'house1').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy - 430);
   180→   180→    this.add.image(cx - 320, cy - 300, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 300);
   181→   181→    this.add.image(cx + 320, cy - 280, 'tower').setScale(0.85).setDepth(DEPTH.ENTITY_BASE + cy - 280);
   182→   182→    this.add.image(cx + 60, cy + 470, 'monastery').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + cy + 470);
   183→   183→    place(cx + 60, cy + 470, 160);
   184→   184→
   185→   185→    // Дома и башни по карте
   186→   186→    for (let i = 0; i < 8; i++) {
   187→   187→      const [x, y] = randPos();
   188→   188→      if (canPlace(x, y, 120)) {
   189→   189→        const k = Phaser.Math.RND.pick(['house1', 'house2', 'house3']);
   190→   190→        this.add.image(x, y, k).setScale(0.75).setDepth(DEPTH.ENTITY_BASE + y);
   191→   191→        place(x, y, 120);
   192→   192→      }
   193→   193→    }
   194→   194→    for (let i = 0; i < 6; i++) {
   195→   195→      const [x, y] = randPos();
   196→   196→      if (canPlace(x, y, 110)) {
   197→   197→        this.add.image(x, y, 'tower').setScale(0.8).setDepth(DEPTH.ENTITY_BASE + y);
   198→   198→        place(x, y, 110);
   199→   199→      }
   200→   200→    }
   201→   201→    // Деревья
   202→   202→    for (let i = 0; i < 70; i++) {
   203→   203→      const [x, y] = randPos();
   204→   204→      if (!canPlace(x, y, 80)) continue;
   205→   205→      const key = rng.frac() < 0.6 ? 'tree1' : 'tree3';
   206→   206→      const sc = 0.8 + rng.frac() * 0.35;
   207→   207→      const t = this.add.sprite(x, y, key).setScale(sc);
   208→   208→      t.play(key === 'tree1' ? 'tree1-sway' : 'tree3-sway');
   209→   209→      if (t.anims.currentAnim) t.setFrame(rng.between(0, 7));
   210→   210→      t.setDepth(DEPTH.ENTITY_BASE + y);
   211→   211→      place(x, y, 80);
   212→   212→    }
   213→   213→    // Камни и кусты
   214→   214→    for (let i = 0; i < 30; i++) {
   215→   215→      const [x, y] = randPos();
   216→   216→      if (!canPlace(x, y, 40)) continue;
   217→   217→      this.add.image(x, y, `rock${rng.between(1, 3)}`).setScale(0.8 + rng.frac() * 0.5)
   218→   218→        .setDepth(DEPTH.ENTITY_BASE + y - 4);
   219→   219→      place(x, y, 40);
   220→   220→    }
   221→   221→    for (let i = 0; i < 40; i++) {
   222→   222→      const [x, y] = randPos();
   223→   223→      if (!canPlace(x, y, 40)) continue;
   224→   224→      const b = this.add.sprite(x, y, 'bushe1').setScale(0.7 + rng.frac() * 0.4);
   225→   225→      b.play('bushe1-sway');
   226→   226→      b.setFrame(rng.between(0, 7));
   227→   227→      b.setDepth(DEPTH.ENTITY_BASE + y - 2);
   228→   228→      place(x, y, 40);
   229→   229→    }
   230→   230→    // Овечки
   231→   231→    for (let i = 0; i < 10; i++) {
   232→   232→      const [x, y] = randPos();
   233→   233→      if (!canPlace(x, y, 50)) continue;
   234→   234→      const sh = this.add.sprite(x, y, 'sheep').setScale(0.8);
   235→   235→      sh.play('sheep-idle');
   236→   236→      sh.setDepth(DEPTH.ENTITY_BASE + y);
   237→   237→      place(x, y, 50);
   238→   238→    }
   239→   239→    // Облака (дрейф вправо, относительный твин с переносом)
   240→   240→    for (let i = 0; i < 10; i++) {
   241→   241→      const [x, y] = [rng.between(0, S), rng.between(0, S)];
   242→   242→      const c = this.add.image(x, y, `cloud${rng.between(1, 3)}`)
   243→   243→        .setAlpha(0.5)
   244→   244→        .setDepth(DEPTH.FX + 50);
   245→   245→      this.tweens.add({
   246→   246→        targets: c,
   247→   247→        x: '+=700',
   248→   248→        duration: rng.between(60, 100) * 1000,
   249→   249→        repeat: -1,
   250→   250→        onRepeat: () => { c.x = -320; },
   251→   251→      });
   252→   252→    }
   253→   253→  }

   296→   296→      moveDir: () => this.lastMove,

   304→   304→    this.hud = new HUD(this, () => this.openPause());

   306→   306→  }

   308→   308→  private setupCollisions(): void {
   309→   309→    this.physics.add.overlap(this.player, this.enemyGroup, (_p, e) => {
   310→   310→      const enemy = e as Enemy;
   311→   311→      this.hurtPlayer(enemy.damage);
   312→   312→    });
   313→   313→
   314→   314→    this.physics.add.overlap(this.projectiles.arrows, this.enemyGroup, (proj, e) => {
   315→   315→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   316→   316→      const enemy = e as Enemy;
   317→   317→      if (!p.active || !enemy.active) return;
   318→   318→      if (p.hitSet?.has(enemy)) return;
   319→   319→      p.hitSet?.add(enemy);
   320→   320→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 90);
   321→   321→      this.projectiles.consumePierce(p);
   322→   322→    });
   323→   323→
   324→   324→    this.physics.add.overlap(this.projectiles.spears, this.enemyGroup, (proj, e) => {
   325→   325→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   326→   326→      const enemy = e as Enemy;
   327→   327→      if (!p.active || !enemy.active) return;
   328→   328→      if (p.hitSet?.has(enemy)) return;
   329→   329→      p.hitSet?.add(enemy);
   330→   330→      this.damageEnemy(enemy, p.dmg, 0, 0, 40);
   331→   331→      this.projectiles.consumePierce(p);
   332→   332→    });
   333→   333→
   334→   334→    this.physics.add.overlap(this.projectiles.knives, this.enemyGroup, (proj, e) => {
   335→   335→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   336→   336→      const enemy = e as Enemy;
   337→   337→      if (!p.active || !enemy.active) return;
   338→   338→      if (p.hitSet?.has(enemy)) return;
   339→   339→      p.hitSet?.add(enemy);
   340→   340→      this.damageEnemy(enemy, p.dmg, enemy.x - this.player.x, enemy.y - this.player.y, 70);
   341→   341→      this.projectiles.consumePierce(p);
   342→   342→    });
   343→   343→
   344→   344→    this.physics.add.overlap(this.player, this.projectiles.enemyArrows, (_p, proj) => {
   345→   345→      const p = proj as unknown as import('../systems/Projectiles').Proj;
   346→   346→      if (!p.active) return;
   347→   347→      this.projectiles.releaseEnemyArrow(p);
   348→   348→      this.hurtPlayer(p.dmg);
   349→   349→    });
   350→   350→  }

   373→   373→  update(time: number, delta: number): void {
   374→   374→    if (!this.started || this.gameEnded) return;
   375→   375→    const dt = Math.min(delta, 50) / 1000;
   376→   376→
   377→   377→    // Пауза
   378→   378→    if (Phaser.Input.Keyboard.JustDown(this.keys.esc) || Phaser.Input.Keyboard.JustDown(this.keys.p)) {
   379→   379→      this.openPause();
   380→   380→      return;
   381→   381→    }
   382→   382→    if (Phaser.Input.Keyboard.JustDown(this.keys.m)) {
   383→   383→      const muted = audio.toggleMute();
   384→   384→      this.hud.announce(muted ? 'ЗВУК ВЫКЛ' : 'ЗВУК ВКЛ', '#ffe9b0');
   385→   385→    }
   386→   386→
   387→   387→    this.elapsed += dt;
   388→   388→
   389→   389→    // Движение игрока
   390→   390→    let vx = 0, vy = 0;
   391→   391→    if (this.keys.a.isDown || this.keys.left.isDown) vx -= 1;
   392→   392→    if (this.keys.d.isDown || this.keys.right.isDown) vx += 1;
   393→   393→    if (this.keys.w.isDown || this.keys.up.isDown) vy -= 1;
   394→   394→    if (this.keys.s.isDown || this.keys.down.isDown) vy += 1;
   395→   395→    const jv = this.joystick.vector;

   598→   598→  private separateEnemies(): void {
   599→   599→    this.sepGrid.clear();
   600→   600→    const cell = 64;
   601→   601→    const px = this.player.x, py = this.player.y;
   602→   602→    const near: Enemy[] = [];
   603→   603→    for (const e of this.enemyGroup.getChildren() as Enemy[]) {
   604→   604→      if (!e.active) continue;
   605→   605→      if (Math.abs(e.x - px) > 900 || Math.abs(e.y - py) > 700) continue;
   606→   606→      near.push(e);
   607→   607→      const key = (Math.floor(e.x / cell) << 16) ^ Math.floor(e.y / cell);
   608→   608→      const arr = this.sepGrid.get(key);
   609→   609→      if (arr) arr.push(e);
   610→   610→      else this.sepGrid.set(key, [e]);
   611→   611→    }
   612→   612→    for (const e of near) {
   613→   613→      const cx = Math.floor(e.x / cell), cy = Math.floor(e.y / cell);
   614→   614→      for (let gx = cx - 1; gx <= cx + 1; gx++) {
   615→   615→        for (let gy = cy - 1; gy <= cy + 1; gy++) {
   616→   616→          const arr = this.sepGrid.get((gx << 16) ^ gy);
   617→   617→          if (!arr) continue;
   618→   618→          for (const o of arr) {
   619→   619→            if (o === e || o.x < e.x) continue; // каждая пара один раз
   620→   620→            const dx = o.x - e.x, dy = o.y - e.y;
   621→   621→            const d = Math.hypot(dx, dy);
   622→   622→            const min = (e.def.bodyRadius + o.def.bodyRadius) * 0.85;
   623→   623→            if (d > 0.001 && d < min) {
   624→   624→              const push = (min - d) / 2;
   625→   625→              const nx = dx / d, ny = dy / d;
   626→   626→              e.x -= nx * push; e.y -= ny * push;
   627→   627→              o.x += nx * push; o.y += ny * push;
   628→   628→            }
   629→   629→          }
   630→   630→        }
   631→   631→      }
   632→   632→    }
   633→   633→  }

   768→   768→      const diff = newMax - this.player.maxHp;

   792→   792→      + Math.floor(this.elapsed / 60) * SCORE.MINUTE

   797→   797→  private victory(): void {
   798→   798→    if (this.gameEnded) return;
   799→   799→    this.score += SCORE.BOSS;
   800→   800→    this.hud.hideBossBar();
   801→   801→    this.hud.announce('ПОБЕДА!!! БОСС ПОВЕРЖЕН!', '#ffd700');
   802→   802→    audio.play('victory');
   803→   803→    this.cameras.main.shake(500, 0.006);
   804→   804→    // Финальный фейерверк
   805→   805→    for (let i = 0; i < 8; i++) {
   806→   806→      this.time.delayedCall(i * 160, () => {
   807→   807→        this.effects.explosion(
   808→   808→          this.player.x + Phaser.Math.Between(-300, 300),
   809→   809→          this.player.y + Phaser.Math.Between(-300, 300), 1.6, 2,
   810→   810→        );
   811→   811→      });
   812→   812→    }
   813→   813→    // Собрать все кристаллы
   814→   814→    this.pickups.vacuumAll(this.player.x, this.player.y);
   815→   815→    this.time.delayedCall(1900, () => this.gameOver(true));
   816→   816→  }
   817→   817→
   818→   818→  private gameOver(victory: boolean): void {
   819→   819→    if (this.gameEnded) return;
   820→   820→    this.gameEnded = true;
   821→   821→    this.spawner.stopped = true;
   822→   822→    this.player.setVelocity(0, 0);
   823→   823→    if (!victory) audio.play('gameover');
   824→   824→    audio.stopMusic();
   825→   825→
   826→   826→    const stats = {
   827→   827→      victory,
   828→   828→      time: Math.floor(this.elapsed),
   829→   829→      kills: this.kills,
   830→   830→      level: this.level,
   831→   831→      gold: this.gold,
   832→   832→      score: this.currentScore(),
   833→   833→      hero: this.heroDef.name,
   834→   834→    };
   835→   835→    this.scene.launch('GameOver', { stats });
   836→   836→    this.scene.pause();
   837→   837→    // Уведомляем React-слой
   838→   838→    if (typeof window !== 'undefined') {
   839→   839→      window.dispatchEvent(new CustomEvent('tss-gameover', { detail: stats }));
   840→   840→    }
   841→   841→  }

   850→   850→  private makeEnemyCtx(): EnemyContext {

   867→   867→        }
