   243→  private createHeroCard(x: number, y: number, heroId: HeroId, cardW: number, cardH: number): void {
   244→    const hero = HEROES[heroId];
   245→    const container = this.add.container(x, y);
   246→    const paper = makePanel(this, 0, 0, 'paper', cardW, cardH);
   247→    container.add(paper);
   248→
   249→    // Живой спрайт героя
   250→    const sprite = this.add.sprite(0, -cardH * 0.27, hero.anims.idle);
   251→    sprite.setScale(hero.scale * 1.05);
   252→    sprite.play(hero.anims.idle);
   253→    container.add(sprite);
   254→
   255→    const name = this.add.text(0, cardH * 0.08, hero.name, {
   256→      fontFamily: FONT, fontSize: '17px', color: '#3b2410',
   257→    }).setOrigin(0.5);
   258→    container.add(name);
   259→
   260→    const title = this.add.text(0, cardH * 0.21, hero.title, {
   261→      fontFamily: FONT, fontSize: '11px', color: '#8a5b12',
   262→    }).setOrigin(0.5);
   263→    container.add(title);
   264→
   265→    const traits = hero.traits.map((t) => `• ${t}`).join('\n');
   266→    const body = this.add.text(0, cardH * 0.30, traits, {
   267→      fontFamily: FONT, fontSize: '10px', color: '#57431f',
   268→      align: 'center', lineSpacing: 5, wordWrap: { width: cardW - 30 },
   269→    }).setOrigin(0.5, 0);
   270→    container.add(body);
   271→
   272→    // Иконка стартового оружия — в углу карточки (не под текстом!)
   273→    const wIcon = this.add.image(cardW / 2 - 26, -cardH / 2 + 26, WEAPONS[hero.startWeapon].icon).setDisplaySize(30, 30);
   274→    container.add(wIcon);
   275→
   276→    // Рамка выделения
   277→    const frame = this.add.graphics();
   278→    container.add(frame);
   279→
   280→    container.setSize(cardW, cardH);
   281→    container.setInteractive({ useHandCursor: true });
   282→    container.on('pointerover', () => {
   283→      if (this.selectedHero !== heroId) {
   284→        this.tweens.add({ targets: container, scale: 1.04, duration: 110 });
   285→      }
   286→    });
   287→    container.on('pointerout', () => {
   288→      if (this.selectedHero !== heroId) {
   289→        this.tweens.add({ targets: container, scale: 1, duration: 110 });
   290→      }
   291→    });
   292→    container.on('pointerdown', () => {
   293→      audio.play('click');
   294→      this.selectHero(heroId);
   295→    });
   296→
   297→    this.heroCards.push({ container, sprite, frame, heroId });
   298→  }

   300→  private createDiffChip(x: number, y: number, w: number, h: number, diffId: DifficultyId): void {
   301→    const def = DIFFICULTIES[diffId];
   302→    const container = this.add.container(x, y);
   303→    const paper = makePanel(this, 0, 0, 'paper', w, h);
   304→    container.add(paper);
   305→
   306→    const isNightmare = diffId === 'nightmare';
   307→    const icon = this.add.text(-w / 2 + 16, 0, isNightmare ? '☠' : '⚔', {
   308→      fontFamily: FONT, fontSize: '16px', color: isNightmare ? '#7a1010' : '#3b2410',
   309→    }).setOrigin(0.5);
   310→    container.add(icon);
   311→
   312→    const name = this.add.text(-w / 2 + 34, -7, def.name.toUpperCase(), {
   313→      fontFamily: FONT, fontSize: '13px', color: isNightmare ? '#7a1010' : '#3b2410',
   314→    }).setOrigin(0, 0.5);
   315→    container.add(name);
   316→
   317→    const tag = this.add.text(-w / 2 + 34, 7, def.tagline, {
   318→      fontFamily: FONT, fontSize: '8px', color: '#8a5b12',
   319→    }).setOrigin(0, 0.5);
   320→    container.add(tag);
   321→
   322→    const frame = this.add.graphics();
   323→    container.add(frame);
   324→
   325→    container.setSize(w, h);
   326→    container.setInteractive({ useHandCursor: true });
   327→    container.on('pointerover', () => {
   328→      if (this.selectedDiff !== diffId) this.tweens.add({ targets: container, scale: 1.04, duration: 100 });
   329→    });
   330→    container.on('pointerout', () => {
   331→      if (this.selectedDiff !== diffId) this.tweens.add({ targets: container, scale: 1, duration: 100 });
   332→    });
   333→    container.on('pointerdown', () => {
   334→      audio.play('click');
   335→      if (this.selectedDiff !== diffId) {
   336→        this.selectedDiff = diffId;
   337→        this.refreshSelection();
   338→      }
   339→    });
   340→
   341→    this.diffChips.push({ container, frame, diffId });
   342→  }
   343→
   344→  private selectHero(heroId: HeroId): void {
   345→    if (this.selectedHero === heroId) return;
   346→    this.selectedHero = heroId;
   347→    this.refreshSelection();
   348→  }
   349→
   350→  private refreshSelection(): void {
   351→    for (const c of this.heroCards) {
   352→      const selected = c.heroId === this.selectedHero;
   353→      c.frame.clear();
   354→      if (selected) {
   355→        c.frame.lineStyle(4, 0xd9a520, 1).strokeRect(-c.container.width / 2 + 4, -c.container.height / 2 + 4, c.container.width - 8, c.container.height - 8);
   356→        c.container.setScale(1.07);
   357→      } else {
   358→        c.container.setScale(1);
   359→      }
   360→    }
   361→    for (const d of this.diffChips) {
   362→      const selected = d.diffId === this.selectedDiff;
   363→      d.frame.clear();
   364→      d.container.setAlpha(selected ? 1 : 0.72);
   365→      if (selected) {
   366→        const w = d.container.width, h = d.container.height;
   367→        d.frame.lineStyle(3, d.diffId === 'nightmare' ? 0xc03028 : 0xd9a520, 1)
   368→          .strokeRect(-w / 2 + 3, -h / 2 + 3, w - 6, h - 6);
   369→        d.container.setScale(1.05);
   370→      } else {
   371→        d.container.setScale(1);
   372→      }
   373→    }
   374→  }
